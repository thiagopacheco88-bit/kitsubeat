    ```typescript
    /**
     * Phase 14 — Theme preference reads/writes (D-08, D-09, D-11).
     *
     * Persistence model: DB column + kb_theme cookie. The cookie is the SSR source
     * of truth on next request (layout.tsx reads it via next/headers cookies());
     * the DB column survives session expiry and replicates across devices for
     * authenticated users. localStorage is NOT used (cookie must be SSR-readable).
     *
     * Validation: server-side enum check + DB CHECK constraint (drizzle/0016).
     */
    const VALID_THEMES = ["system", "light", "dark"] as const;
    type ThemePreference = (typeof VALID_THEMES)[number];

    export async function setThemePreference(
      userId: string,
      value: ThemePreference
    ): Promise<void> {
      if (!userId) throw new Error("userId is required");
      if (!VALID_THEMES.includes(value)) {
        throw new Error(`themePreference must be one of: ${VALID_THEMES.join(", ")}`);
      }

      // DB write — upsert seeds row if absent. Mirrors updateUserPrefs idiom (lines 97-112).
      await db
        .insert(users)
        .values({ id: userId, themePreference: value })
        .onConflictDoUpdate({
          target: users.id,
          set: { themePreference: value, updated_at: new Date() },
        });

      // Cookie write — D-08 specifics: kb_theme, 1-year expiry, SameSite=Lax, NOT HttpOnly
      // (client must read for instant toggle without server round-trip).
      // Pitfall 2: cookies() is async in Next 15; await it.
      const { cookies } = await import("next/headers");
      const c = await cookies();
      c.set("kb_theme", value, {
        maxAge: 60 * 60 * 24 * 365, // 1 year in seconds
        sameSite: "lax",
        httpOnly: false, // D-08 — client-readable for optimistic toggle
        path: "/",
      });
    }

    export async function getThemePreference(userId: string): Promise<ThemePreference> {
      if (!userId) return "system";
      const rows = await db
        .select({ themePreference: users.themePreference })
        .from(users)
        .where(eq(users.id, userId))
        .limit(1);
      return (rows[0]?.themePreference as ThemePreference) ?? "system";
    }
    ```

Use Lucide icons (already in node_modules per other components — verify with grep) or inline SVG (D-discretion). Use `<Button variant="ghost" size="sm">` for visual consistency.
    ```typescript
    "use client";

    import { useEffect, useState, useTransition } from "react";
    import { Sun, Moon, Monitor } from "lucide-react"; // verify lucide-react is installed; if not, use inline SVG
    import { Button } from "./Button";
    import { setThemePreference } from "@/app/actions/userPrefs";
    import { useUser } from "@clerk/nextjs";

    type ThemePref = "system" | "light" | "dark";
    const ORDER: readonly ThemePref[] = ["system", "light", "dark"] as const;

    function readCookie(): ThemePref {
      if (typeof document === "undefined") return "system";
      const m = document.cookie.match(/kb_theme=(system|light|dark)/);
      return (m?.[1] as ThemePref) ?? "system";
    }

    function resolveSystem(value: ThemePref): "light" | "dark" {
      if (value === "light" || value === "dark") return value;
      return window.matchMedia("(prefers-color-scheme: dark)").matches ? "dark" : "light";
    }

    export function ThemeToggle() {
      const [pref, setPref] = useState<ThemePref>("system");
      const [isPending, startTransition] = useTransition();
      const { user } = useUser();

      useEffect(() => { setPref(readCookie()); }, []);

      const cycle = () => {
        const next = ORDER[(ORDER.indexOf(pref) + 1) % ORDER.length];
        // Optimistic update — apply data-theme BEFORE server action returns (per D-10)
        const resolved = resolveSystem(next);
        document.documentElement.setAttribute("data-theme", resolved);
        // Optimistic cookie write so a refresh while server action is in flight stays consistent
        document.cookie = `kb_theme=${next}; max-age=${60 * 60 * 24 * 365}; path=/; samesite=lax`;
        setPref(next);

        // Server action async — DB column write for authed users; cookie write is redundant but defensive
        if (user?.id) {
          startTransition(async () => {
            try {
              await setThemePreference(user.id, next);
            } catch (e) {
              // On failure: revert optimistic state. Toast TBD — for v1 just console.error.
              console.error("setThemePreference failed:", e);
              const prev = pref;
              document.documentElement.setAttribute("data-theme", resolveSystem(prev));
              document.cookie = `kb_theme=${prev}; max-age=${60 * 60 * 24 * 365}; path=/; samesite=lax`;
              setPref(prev);
            }
          });
        }
      };

      const Icon = pref === "system" ? Monitor : pref === "light" ? Sun : Moon;
      const label = `Theme: ${pref}. Click to change.`;

      return (
        <Button
          variant="ghost"
          size="sm"
          onClick={cycle}
          aria-label={label}
          title={label}
          disabled={isPending}
          className="!min-h-[44px] !min-w-[44px] !p-2"
        >
          <Icon className="h-5 w-5" aria-hidden="true" />
        </Button>
      );
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Cookie input → server action | Cookie value can be tampered client-side; server action must validate before DB write |
| Server action input → DB | Drizzle parameterizes; CHECK constraint also enforces enum at DB layer |
| Inline `<script>` in `<head>` | First-class XSS surface — interpolated string would bypass React's escaping |
| Server action invocation | Next.js server actions ship with built-in CSRF protection (action ID + origin check) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-03-01 | XSS / Tampering | Inline `<script>` in layout.tsx `<head>` | mitigate | The `__html` field uses a LITERAL TEMPLATE STRING. NO USER INPUT EVER INTERPOLATED. Verified by grep: `grep '\${' src/app/layout.tsx` should match nothing inside the script tag. The script's regex (`kb_theme=(system\|light\|dark)`) constrains the cookie value to a 3-element enum BEFORE setting the data-theme attribute — even a maliciously-set `kb_theme=<script>` cookie value falls back to 'system' (the regex doesn't match) |
| T-14-03-02 | Tampering | Cookie tampering — user sets `kb_theme=<malicious>` | mitigate | Three layers of defense: (a) inline script's regex match constrains to enum (any other value falls through to 'system'); (b) server action's `VALID_THEMES` check throws on invalid input before DB write; (c) DB CHECK constraint enforces enum at the data layer. Defense-in-depth — three layers must all be bypassed for an invalid value to land |
| T-14-03-03 | Spoofing / Authorization | Server action called with someone else's userId | mitigate | `setThemePreference` accepts userId as the FIRST positional argument. The header ThemeToggle and ProfileForm both source userId from `useUser()` from Clerk (server-validated session). The server action does NOT trust an arbitrary userId — but the API surface is exposed via "use server" so Plan 14-04 / future audit (Phase 16 D-29 server-action authz audit) should add a `requireMatchingSession(userId)` gate. For Phase 14: ACCEPT the risk because (a) the worst case is a user's theme is wrong on next page load — no PII or money at stake; (b) Phase 16 has the broader server-action authz audit |
| T-14-03-04 | CSRF | Server action invoked from external origin | mitigate | Next.js server actions ship with CSRF protection via the action-ID header + origin check (per Next 15 docs). No additional code needed |
| T-14-03-05 | SQL Injection | New `theme_preference` column query | mitigate | Drizzle ORM uses parameterized queries (`.values({ themePreference: value })` and `eq(users.id, userId)`). String values never concatenated into SQL. Verified by reading the upsert pattern at userPrefs.ts:97-112 |
| T-14-03-06 | Information Disclosure | `kb_theme` cookie readable by other scripts | accept | Per D-08 the cookie is intentionally NOT HttpOnly so the client can read it for instant toggle. The cookie value is `'system' \| 'light' \| 'dark'` — no PII, no auth tokens, no session secrets. Worst case: another script reads "this user prefers dark theme". Acceptable risk for the UX win of optimistic toggle |
| T-14-03-07 | Denial of Service | RSC re-render storm from cookie write | accept | Pitfall 10: cookies().set() inside server action triggers Next's RSC re-render. The optimistic client-side data-theme set is immediate; the server's RSC re-render arrives later but renders the same state. No flicker (verified in tests/e2e/theme-toggle.spec.ts test 6) |
