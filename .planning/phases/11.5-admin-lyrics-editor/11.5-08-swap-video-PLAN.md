<objective>
Implement YouTube video swap end-to-end: admin enters a new URL + selects which pipeline steps to re-run + writes a reason; server action validates the URL, writes a `song_video_history` audit row, sets `song_versions.pipeline_status='rerun_in_progress'` (guarded), and spawns `scripts/seed/swap-video-pipeline.ts` as a detached child process. The orchestrator runs per-step (audio download → Demucs+WhisperX → lesson regen → DB insert) writing progress to `pipeline_step` so the editor's poller can show transitions. On failure, retry-from-failed-step (D-12); on second failure, `quality_status='flagged_unfixable'`.

Purpose: SPEC #19/#20/#21 — wrong-song matched and broken videos must be fixable. Today the only path is manual DB edit + script rerun. This plan gives admin a UI to swap and observe the rerun. Crucially: pipeline_status='rerun_in_progress' hides the song from the public catalog (Plan 09's responsibility) so users don't see a broken transitional state.

Output: Orchestrator script with per-step status writes + retry-from-failed-step logic, server action that spawns it, polling API endpoint, modal with URL/steps/reason inputs, status poller component with retry button, two integration tests for status writes and failure handling.

```typescript
#!/usr/bin/env tsx
/**
 * Phase 11.5 D-09: YouTube swap orchestrator.
 *
 * Usage:
 *   tsx scripts/seed/swap-video-pipeline.ts \
 *     --slug=<slug> \
 *     --new-youtube-id=<11-char-id> \
 *     --steps=audio,whisper,lesson \
 *     [--resume-from=<step>]
 *
 * Per-step DB writes:
 *   BEFORE step: UPDATE song_versions SET pipeline_step=<name>
 *   ON success of last step (if --steps includes 'lesson'): UPDATE pipeline_status='idle', pipeline_step=NULL
 *   ON failure (after retry exhausted): UPDATE pipeline_status='rerun_failed', pipeline_step=<failed>; UPDATE songs SET quality_status='flagged_unfixable' if retry was used
 *
 * Per RESEARCH §"Pre-DB script bootstrap": uses neon-serverless Pool (not neon-http).
 */

import { config } from "dotenv";
config({ path: ".env.local" });

import { execSync } from "node:child_process";
import { Pool } from "@neondatabase/serverless";
import { drizzle } from "drizzle-orm/neon-serverless";
import { sql } from "drizzle-orm";

interface CliArgs {
  slug: string;
  newYoutubeId: string;
  steps: Set<"audio" | "whisper" | "lesson" | "lyrics" | "insert">;
  resumeFrom: string | null;
  isRetry: boolean;  // true when --resume-from is provided
}

function parseArgs(): CliArgs {
  const argv = process.argv.slice(2);
  const get = (name: string) => {
    const a = argv.find((s) => s.startsWith(`--${name}=`));
    return a ? a.slice(name.length + 3) : null;
  };
  const slug = get("slug");
  const newYoutubeId = get("new-youtube-id");
  const stepsRaw = get("steps") ?? "audio,whisper,lesson";
  const resumeFrom = get("resume-from");

  if (!slug) throw new Error("--slug is required");
  if (!newYoutubeId) throw new Error("--new-youtube-id is required");
  if (!/^[a-zA-Z0-9_-]{11}$/.test(newYoutubeId)) {
    throw new Error("--new-youtube-id must be exactly 11 chars matching ^[a-zA-Z0-9_-]+$");
  }

  const steps = new Set(stepsRaw.split(",").map((s) => s.trim()) as CliArgs["steps"] extends Set<infer T> ? T[] : never);

  return { slug, newYoutubeId, steps, resumeFrom, isRetry: !!resumeFrom };
}

interface PipelineStep {
  name: "audio" | "whisper" | "lesson" | "lyrics" | "insert";
  command: string;
}

function makeSteps(args: CliArgs): PipelineStep[] {
  // Concrete commands depend on existing scripts. The placeholder shapes here:
  return [
    { name: "audio",   command: `tsx scripts/seed/02-fetch-audio.ts --slug=${args.slug} --youtube-id=${args.newYoutubeId}` },
    { name: "whisper", command: `python scripts/seed/04-extract-timing.py --slug=${args.slug}` },
    { name: "lesson",  command: `npm run generate:lessons -- --slug=${args.slug}` },
    { name: "lyrics",  command: `tsx scripts/seed/04b-backfill-whisper-lyrics.ts --slug=${args.slug}` },
    { name: "insert",  command: `tsx scripts/seed/05-insert-db.ts --slug=${args.slug}` },
  ].filter((s) => args.steps.has(s.name));
}

async function main() {
  const args = parseArgs();
  const pool = new Pool({ connectionString: process.env.DATABASE_URL });
  const db = drizzle(pool);

  // Look up song_version_id by slug (Phase 11.5 caller uses songVersionId; this script
  // uses slug for compatibility with existing per-slug filter pattern)
  const r = await db.execute<{ id: string }>(sql`
    SELECT sv.id::text AS id
    FROM song_versions sv
    INNER JOIN songs s ON s.id = sv.song_id
    WHERE s.slug = ${args.slug}
    LIMIT 1
  `);
  const rows = Array.isArray(r) ? r : (r.rows ?? []);
  if (rows.length === 0) {
    console.error(`[FAIL] no song_version found for slug=${args.slug}`);
    process.exit(1);
  }
  const songVersionId = rows[0].id;

  const steps = makeSteps(args);
  if (steps.length === 0) {
    console.error("no steps to run");
    process.exit(1);
  }

  let resumeIdx = 0;
  if (args.resumeFrom) {
    const idx = steps.findIndex((s) => s.name === args.resumeFrom);
    if (idx >= 0) resumeIdx = idx;
  }

  for (let i = resumeIdx; i < steps.length; i++) {
    const step = steps[i];
    await db.execute(sql`UPDATE song_versions SET pipeline_step = ${step.name}, pipeline_started_at = COALESCE(pipeline_started_at, now()) WHERE id = ${songVersionId}::uuid`);
    console.log(`\n[RUN] ${step.name}: ${step.command}`);
    try {
      execSync(step.command, { stdio: "inherit" });
      console.log(`[DONE] ${step.name}`);
    } catch (err) {
      const errMsg = (err as Error).message.replace(/'/g, "''");
      const ts = new Date().toISOString();
      console.error(`[FAIL] ${step.name}: ${errMsg}`);

      if (args.isRetry) {
        // D-12: 2nd failure → flagged_unfixable
        await db.execute(sql`
          UPDATE songs
            SET quality_status = 'flagged_unfixable',
                quality_notes = COALESCE(quality_notes || E'\n', '') ||
                  ${`Pipeline failed at ${step.name}: ${errMsg.slice(0, 200)} (${ts})`}
            WHERE id IN (SELECT song_id FROM song_versions WHERE id = ${songVersionId}::uuid)
        `);
        await db.execute(sql`UPDATE song_versions SET pipeline_status = 'rerun_failed', pipeline_step = ${step.name} WHERE id = ${songVersionId}::uuid`);
      } else {
        await db.execute(sql`UPDATE song_versions SET pipeline_status = 'rerun_failed', pipeline_step = ${step.name} WHERE id = ${songVersionId}::uuid`);
      }
      await pool.end();
      process.exit(2);
    }
  }

  // All steps succeeded
  if (args.steps.has("lesson")) {
    // Plan 07's publish flow OR scripts/seed/05-insert-db.ts is responsible for writing the new
    // source='auto' lyrics_versions row. The orchestrator's role here is to flip status → idle.
    await db.execute(sql`UPDATE song_versions SET pipeline_status = 'idle', pipeline_step = NULL WHERE id = ${songVersionId}::uuid`);
  } else {
    // Pitfall 13: no lesson step ran → do NOT write a new auto version, just clear status
    await db.execute(sql`UPDATE song_versions SET pipeline_status = 'idle', pipeline_step = NULL WHERE id = ${songVersionId}::uuid`);
  }

  console.log(`\n[OK] swap-video-pipeline complete for slug=${args.slug}`);
  await pool.end();
  process.exit(0);
}

main().catch((err) => {
  console.error("[FATAL]", err);
  process.exit(1);
});
```

 * Phase 11.5 SPEC #19/#20/#21 + D-09/D-10/D-11/D-12: YouTube swap server action.

 * Validates URL → audit log insert → guarded status flip → spawn detached orchestrator.
 *

export const runtime = "nodejs";

import { spawn } from "node:child_process";
import path from "node:path";
import { db } from "@/lib/db";
import { sql } from "drizzle-orm";
import { requireAdminUser } from "@/lib/admin/require-admin";

const ALLOWED_HOSTS = new Set(["youtube.com", "www.youtube.com", "m.youtube.com", "youtu.be"]);
const VIDEO_ID_RE = /^([a-zA-Z0-9_-]{11})$/;

export interface SwapVideoInput {
  songVersionId: string;
  slug: string;
  newYoutubeUrl: string;
  steps: ("audio" | "whisper" | "lesson" | "lyrics" | "insert")[];

export type SwapVideoOutput =
  | { ok: true; jobStartedAt: string; newYoutubeId: string }
  | { ok: false; error: "invalid_url" | "pipeline_already_in_progress" | "song_not_found" | string };

export async function swapVideo(input: SwapVideoInput): Promise<SwapVideoOutput> {
  const admin = await requireAdminUser();

  // Step 1: validate URL (T-11.5-05)
  let url: URL;
  try {
    url = new URL(input.newYoutubeUrl);
  } catch {
    return { ok: false, error: "invalid_url" };
  }
  if (!ALLOWED_HOSTS.has(url.hostname)) {
    return { ok: false, error: "invalid_url" };
  }

  // Extract 11-char video ID — supports v=ID and youtu.be/ID
  let candidate: string | null = null;
  if (url.searchParams.has("v")) candidate = url.searchParams.get("v");
  else if (url.hostname === "youtu.be") candidate = url.pathname.slice(1);
  else {
    // /watch path or /embed/<id> path
    const m = url.pathname.match(/\/embed\/([a-zA-Z0-9_-]+)/);
    if (m) candidate = m[1];
  }
  if (!candidate || !VIDEO_ID_RE.test(candidate)) {
    return { ok: false, error: "invalid_url" };
  }
  const newYoutubeId = candidate;

  // Step 2: read current youtube_id for audit log
  const cur = await db.execute<{ youtube_id: string | null }>(sql`
    SELECT youtube_id FROM song_versions WHERE id = ${input.songVersionId}::uuid LIMIT 1
  `);
  const curRows = Array.isArray(cur) ? cur : (cur.rows ?? []);
  if (curRows.length === 0) return { ok: false, error: "song_not_found" };
  const oldYoutubeId = curRows[0].youtube_id;

  // Step 3: guarded status flip + audit log insert
  const updated = await db.execute<{ id: string }>(sql`
    UPDATE song_versions
       SET youtube_id = ${newYoutubeId},
           pipeline_status = 'rerun_in_progress',
           pipeline_step = 'queued',
           pipeline_started_at = now()
     WHERE id = ${input.songVersionId}::uuid
       AND pipeline_status = 'idle'
     RETURNING id::text AS id
  `);
  const updatedRows = Array.isArray(updated) ? updated : (updated.rows ?? []);
  if (updatedRows.length === 0 && !input.isRetry) {
    return { ok: false, error: "pipeline_already_in_progress" };
  }

  // For retry, status was already 'rerun_failed' — flip back to in_progress without guard
  if (input.isRetry) {
    await db.execute(sql`
      UPDATE song_versions
         SET pipeline_status = 'rerun_in_progress',
             pipeline_step = ${input.resumeFrom ?? 'queued'},
             pipeline_started_at = now()
       WHERE id = ${input.songVersionId}::uuid
    `);
  }

  await db.execute(sql`
    INSERT INTO song_video_history (song_version_id, old_youtube_id, new_youtube_id, changed_by, reason)
    VALUES (${input.songVersionId}::uuid, ${oldYoutubeId}, ${newYoutubeId}, ${admin.id}, ${input.reason})
  `);

  // Step 4: spawn orchestrator detached
  const stepsArg = input.steps.join(",");
  const args = [
    "tsx",
    path.resolve(process.cwd(), "scripts/seed/swap-video-pipeline.ts"),
    `--slug=${input.slug}`,
    `--new-youtube-id=${newYoutubeId}`,
    `--steps=${stepsArg}`,
  ];
  if (input.resumeFrom) args.push(`--resume-from=${input.resumeFrom}`);

  const child = spawn("npx", args, {
    detached: true,
    stdio: "ignore",
    env: process.env,
    cwd: process.cwd(),
  });
  child.unref();  // allow dev server to exit independently

  return { ok: true, jobStartedAt: new Date().toISOString(), newYoutubeId };
}

"use client";

import { useEffect, useState } from "react";

interface Props {
  songVersionId: string;
  initialStatus: "idle" | "rerun_in_progress" | "rerun_failed";
  initialStep: string | null;
  onRetry: (resumeFrom: string | null) => void;
}

interface PollResponse {
  pipeline_status: "idle" | "rerun_in_progress" | "rerun_failed";
  pipeline_step: string | null;
  pipeline_started_at: string | null;
  quality_status: "active" | "flagged_wrong_song" | "flagged_unfixable";
  quality_notes: string | null;
}

const POLL_INTERVAL_MS = 3000;

export default function PipelineStatusPoller({ songVersionId, initialStatus, initialStep, onRetry }: Props) {
  const [state, setState] = useState<PollResponse>({
    pipeline_status: initialStatus,
    pipeline_step: initialStep,
    pipeline_started_at: null,
    quality_status: "active",
    quality_notes: null,
  });

  useEffect(() => {
    let cancelled = false;
    if (state.pipeline_status === "idle") return;

    async function poll() {
      try {
        const r = await fetch(`/api/admin/song/${songVersionId}/pipeline-status`);
        if (!r.ok) return;
        const data = (await r.json()) as PollResponse;
        if (cancelled) return;
        setState(data);
      } catch { /* network blip — keep polling */ }
    }

    const id = setInterval(poll, POLL_INTERVAL_MS);
    return () => { cancelled = true; clearInterval(id); };
  }, [songVersionId, state.pipeline_status]);

  if (state.pipeline_status === "idle" && state.quality_status === "active") return null;

  return (
    <div data-testid="pipeline-status" style={{ marginTop: "12px", padding: "10px", border: "1px solid #fcd34d", background: "#fef3c7", borderRadius: "4px" }}>
      <p style={{ margin: 0, fontSize: "12px", color: "#92400e", fontWeight: 600 }}>
        Pipeline status: {state.pipeline_status}{state.pipeline_step ? ` (step: ${state.pipeline_step})` : ""}
      </p>
      {state.pipeline_started_at && (
        <p style={{ margin: "4px 0 0 0", fontSize: "11px", color: "#92400e" }}>
          Started: {new Date(state.pipeline_started_at).toLocaleString()}
        </p>
      )}
      {state.quality_status !== "active" && (
        <p style={{ margin: "4px 0 0 0", fontSize: "11px", color: "#dc2626" }}>
          Quality: {state.quality_status}
          {state.quality_notes ? `\n${state.quality_notes}` : ""}
        </p>
      )}
      {state.pipeline_status === "rerun_failed" && (
        <button
          type="button"
          onClick={() => onRetry(state.pipeline_step)}
          data-testid="pipeline-retry"
          style={{ marginTop: "8px", padding: "4px 10px", fontSize: "12px", border: "1px solid #6366f1", borderRadius: "4px", background: "#6366f1", color: "#fff", cursor: "pointer" }}
        >
          Retry from {state.pipeline_step ?? "start"}
        </button>
      )}
    </div>
  );
}
