```typescript
/**
 * scripts/seed/20-seed-anime-vocab.ts
 *
 * Seeds 760 anime vocabulary words from .planning/anime-vocab/*.json into:
 *   1. vocabulary_items (ON CONFLICT DO UPDATE — idempotent)
 *   2. anime_vocab_catalog (ON CONFLICT DO NOTHING — idempotent)
 *
 * Run: npx tsx scripts/seed/20-seed-anime-vocab.ts
 */

import fs from "fs";
import path from "path";
import { db } from "@/lib/db/index";
import { vocabularyItems, animeVocabCatalog } from "@/lib/db/schema";
import { sql } from "drizzle-orm";

// ── JSON file structure ──────────────────────────────────────────────────────

interface AnimeVocabWord {
  surface: string;    // maps to vocabulary_items.dictionary_form
  reading: string;
  romaji: string;
  meanings: { en: string; "pt-BR": string; es: string };
  jlpt_level: string | null;
  category: string;
  context: string;    // maps to anime_vocab_catalog.context_note
}

interface AnimeVocabFile {
  anime: string;      // slug, e.g. "one-piece"
  label: string;      // display name, e.g. "One Piece"
  words: AnimeVocabWord[];
}

// ── JLPT sort order ──────────────────────────────────────────────────────────

const JLPT_ORDER: Record<string, number> = { N5: 0, N4: 1, N3: 2, N2: 3, N1: 4 };
const jlptRank = (level: string | null): number =>
  level ? (JLPT_ORDER[level] ?? 4) : 5;

// ── Slug files ───────────────────────────────────────────────────────────────

const SLUGS = [
  "one-piece",
  "naruto",
  "bleach",
  "fullmetal-alchemist",
  "attack-on-titan",
  "sword-art-online",
];

const VOCAB_DIR = path.join(process.cwd(), ".planning", "anime-vocab");

async function main() {
  let totalVocabUpserted = 0;
  let totalCatalogUpserted = 0;

  for (const slug of SLUGS) {
    const filePath = path.join(VOCAB_DIR, `${slug}.json`);

    if (!fs.existsSync(filePath)) {
      console.warn(`[WARN] Missing file: ${filePath} — skipping`);
      continue;
    }

    const raw = fs.readFileSync(filePath, "utf-8");
    const data: AnimeVocabFile = JSON.parse(raw);
    const words = data.words;

    console.log(`\n[${slug}] Processing ${words.length} words...`);

    // Sort words by JLPT rank first, then by original array index for stable ordering
    const sorted = words
      .map((w, i) => ({ word: w, originalIndex: i }))
      .sort((a, b) => {
        const rankDiff = jlptRank(a.word.jlpt_level) - jlptRank(b.word.jlpt_level);
        return rankDiff !== 0 ? rankDiff : a.originalIndex - b.originalIndex;
      });

    for (let i = 0; i < sorted.length; i++) {
      const { word } = sorted[i];

      // 1. Upsert into vocabulary_items
      //    ON CONFLICT (dictionary_form, reading) DO UPDATE to refresh meaning/romaji
      const [vocabRow] = await db
        .insert(vocabularyItems)
        .values({
          dictionary_form: word.surface,
          reading: word.reading,
          romaji: word.romaji,
          part_of_speech: "noun",   // default — ~85% correct; enrichment can fix later
          jlpt_level: (word.jlpt_level as "N5" | "N4" | "N3" | "N2" | "N1" | null) ?? null,
          meaning: word.meanings,
          is_katakana_loanword: false,
          admin_flagged: false,
        })
        .onConflictDoUpdate({
          target: sql`(dictionary_form, reading)`,
          set: {
            romaji: word.romaji,
            meaning: word.meanings,
            // Do NOT overwrite jlpt_level or part_of_speech if already enriched
          },
        })
        .returning({ id: vocabularyItems.id });

      totalVocabUpserted++;

      // 2. Upsert into anime_vocab_catalog
      //    ON CONFLICT DO NOTHING — idempotent; display_order and context_note are stable
      await db
        .insert(animeVocabCatalog)
        .values({
          anime_slug: slug,
          vocab_item_id: vocabRow.id,
          category: word.category,
          display_order: i,   // i is the sorted position (JLPT rank + original index tiebreak)
          context_note: word.context,
        })
        .onConflictDoNothing();

      totalCatalogUpserted++;
    }

    console.log(`[${slug}] Done — ${words.length} words processed`);
  }

  console.log(`\n✓ Seed complete`);
  console.log(`  vocabulary_items upserted: ${totalVocabUpserted}`);
  console.log(`  anime_vocab_catalog rows: ${totalCatalogUpserted}`);

  process.exit(0);
}

main().catch((err) => {
  console.error("Seed failed:", err);
  process.exit(1);
});
```
