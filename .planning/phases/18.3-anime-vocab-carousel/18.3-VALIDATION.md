### 1. `src/lib/exercises/__tests__/carousel-generator.test.ts`

Covers: CULT-01, CULT-02 (mock), CULT-04

```typescript
import { describe, it, expect } from "vitest";
import { buildAnimeCarouselQuestions } from "../carousel-generator";
import type { AnimeVocabItem } from "@/lib/db/queries";

// Minimal fixture factory
function makeWord(overrides: Partial<AnimeVocabItem> = {}): AnimeVocabItem {
  return {
    vocab_item_id: "aaaaaaaa-0000-0000-0000-000000000001",
    surface: "海賊",
    reading: "かいぞく",
    romaji: "kaizoku",
    meaning: { en: "pirate", "pt-BR": "pirata", es: "pirata" },
    jlpt_level: "N3",
    category: "pirate",
    context_note: "Luffy wants to be King of the Pirates.",
    display_order: 0,
    mastery_state: 0,
    ...overrides,
  };
}

describe("buildAnimeCarouselQuestions", () => {
  it("returns empty array for empty input", () => {
    expect(buildAnimeCarouselQuestions([], "en", 10)).toEqual([]);
  });

  it("returns at most cap questions", () => {
    // 50 words × 2 types = 100 candidates, cap = 10
    const words = Array.from({ length: 50 }, (_, i) =>
      makeWord({ vocab_item_id: `aaaaaaaa-0000-0000-0000-${String(i).padStart(12, "0")}`, surface: `word${i}` })
    );
    const result = buildAnimeCarouselQuestions(words, "en", 10);
    expect(result.length).toBe(10);
  });

  it("produces only vocab_meaning and meaning_vocab types", () => {
    const words = Array.from({ length: 10 }, (_, i) =>
      makeWord({ vocab_item_id: `aaaaaaaa-0000-0000-0000-${String(i).padStart(12, "0")}`, surface: `word${i}` })
    );
    const result = buildAnimeCarouselQuestions(words, "en", 20);
    const types = new Set(result.map((q) => q.type));
    expect(types.has("vocab_meaning")).toBe(true);
    expect(types.has("meaning_vocab")).toBe(true);
    // No other types
    for (const t of types) {
      expect(["vocab_meaning", "meaning_vocab"]).toContain(t);
    }
  });

  it("each question has correctAnswer, distractors (3), and vocabItemId", () => {
    const words = Array.from({ length: 10 }, (_, i) =>
      makeWord({ vocab_item_id: `aaaaaaaa-0000-0000-0000-${String(i).padStart(12, "0")}`, surface: `word${i}` })
    );
    const result = buildAnimeCarouselQuestions(words, "en", 20);
    for (const q of result) {
      expect(q.correctAnswer).toBeTruthy();
      expect(q.distractors).toHaveLength(3);
      expect(q.vocabItemId).toBeTruthy();
    }
  });

  it("returns at most 2×words.length questions when words < cap/2", () => {
    // 3 words × 2 types = 6 candidates max, cap = 20
    const words = Array.from({ length: 3 }, (_, i) =>
      makeWord({ vocab_item_id: `aaaaaaaa-0000-0000-0000-${String(i).padStart(12, "0")}`, surface: `w${i}` })
    );
    const result = buildAnimeCarouselQuestions(words, "en", 20);
    expect(result.length).toBeLessThanOrEqual(6);
  });

  // CULT-04: category filtering (caller filters words array before calling)
  it("respects filtered word set — category filter on caller side reduces question pool", () => {
    const pirateWords = Array.from({ length: 5 }, (_, i) =>
      makeWord({ vocab_item_id: `aaaaaaaa-0000-0000-0000-${String(i).padStart(12, "0")}`, category: "pirate", surface: `p${i}` })
    );
    const nakama = Array.from({ length: 5 }, (_, i) =>
      makeWord({ vocab_item_id: `bbbbbbbb-0000-0000-0000-${String(i).padStart(12, "0")}`, category: "nakama", surface: `n${i}` })
    );
    const allWords = [...pirateWords, ...nakama];
    const filtered = allWords.filter((w) => w.category === "pirate");
    const result = buildAnimeCarouselQuestions(filtered, "en", 20);
    // All questions reference pirate-category word IDs only
    const ids = new Set(result.map((q) => q.vocabItemId));
    for (const id of ids) {
      expect(pirateWords.map((w) => w.vocab_item_id)).toContain(id);
    }
  });
});
```

Run command: `npm run test:unit -- carousel-generator`

### 2. `tests/integration/anime-vocab.test.ts`

Covers: CULT-01 (DB query shape)

This test requires a live DB connection. It should be skipped in CI if `DATABASE_URL` is not set.

```typescript
import { describe, it, expect, beforeAll } from "vitest";
import { db } from "@/lib/db/index";
import { animeVocabCatalog, vocabularyItems } from "@/lib/db/schema";
import { eq } from "drizzle-orm";

const SKIP = !process.env.DATABASE_URL;

describe.skipIf(SKIP)("getAnimeVocabBySlug integration", () => {
  it("returns vocab rows for one-piece after seeding", async () => {
    const rows = await db
      .select()
      .from(animeVocabCatalog)
      .where(eq(animeVocabCatalog.anime_slug, "one-piece"))
      .limit(5);
    expect(rows.length).toBeGreaterThan(0);
    expect(rows[0].anime_slug).toBe("one-piece");
    expect(rows[0].vocab_item_id).toBeTruthy();
  });

  it("JOIN with vocabulary_items returns meaning and romaji", async () => {
    const rows = await db
      .select({
        anime_slug: animeVocabCatalog.anime_slug,
        category: animeVocabCatalog.category,
        dictionary_form: vocabularyItems.dictionary_form,
        romaji: vocabularyItems.romaji,
        meaning: vocabularyItems.meaning,
      })
      .from(animeVocabCatalog)
      .innerJoin(vocabularyItems, eq(animeVocabCatalog.vocab_item_id, vocabularyItems.id))
      .where(eq(animeVocabCatalog.anime_slug, "one-piece"))
      .limit(3);
    expect(rows.length).toBeGreaterThan(0);
    for (const row of rows) {
      expect(row.romaji).toBeTruthy();
      expect(row.meaning).toBeTruthy();
    }
  });
});
```

Run command: `npm run test:integration -- anime-vocab`
