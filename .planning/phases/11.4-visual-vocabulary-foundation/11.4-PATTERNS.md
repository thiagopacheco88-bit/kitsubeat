**Frequency-rank query (NEW pattern — RESEARCH Pattern 6, NO direct in-repo precedent for `vocab_global` JOIN; closest is `src/lib/db/queries.ts:594-610`):**
```typescript
const db = getDb();
await refreshVocabGlobal();              // Pitfall 2 — defensive, CONCURRENTLY-safe, <1s

const rows = await db.execute(sql`
  SELECT
    vi.id            AS vocab_item_id,
    vi.dictionary_form,
    vi.reading,
    vi.romaji,
    vi.meaning,
    vi.part_of_speech,
    vi.jlpt_level::text AS jlpt_level,
    COUNT(*)::int    AS appearance_count
  FROM vocab_global vg
  JOIN vocabulary_items vi ON vi.id = vg.vocab_item_id
  WHERE vi.part_of_speech IN ('noun', 'verb')
    AND vi.image_url IS NULL                    -- skip already-curated rows
  GROUP BY vi.id, vi.dictionary_form, vi.reading, vi.romaji, vi.meaning, vi.part_of_speech, vi.jlpt_level
  ORDER BY appearance_count DESC, vi.dictionary_form ASC
  LIMIT 50
`);
const raw = Array.isArray(rows) ? rows : (rows.rows ?? []);   // Neon HTTP vs node-pg shape
```

```typescript
import { vocabularyItems } from "../../src/lib/db/schema.js";
// ... env loading + getDb() identical to 11-enrich-vocab.ts header ...

const URL_PATTERN = /^https:\/\/images\.unsplash\.com\/.+/;
const TSV_PATH = "data/vocab-images-staging.tsv";

interface TsvRow {
  vocab_item_id: string;
  dictionary_form: string;
  image_url: string;
}

const lines = readFileSync(TSV_PATH, "utf-8")
  .split(/\r?\n/)
  .filter((line) => line.trim() && !line.startsWith("#"));

const tsvRows: TsvRow[] = lines.map((line) => {
  const [vocab_item_id, dictionary_form, , , , , image_url] = line.split("\t");
  return { vocab_item_id, dictionary_form, image_url: (image_url ?? "").trim() };
});

const db = getDb();
let loaded = 0, skipped = 0, invalid = 0;

for (const row of tsvRows) {
  if (!row.image_url) {
    skipped++;
    continue;
  }
  if (!URL_PATTERN.test(row.image_url)) {
    console.warn(
      `[invalid] ${row.dictionary_form} (${row.vocab_item_id}): URL does not match Unsplash CDN pattern: ${row.image_url}`
    );
    invalid++;
    continue;
  }
  // Per-row commit — partial progress survives crashes (matches 11-enrich-vocab.ts:152-158)
  await db
    .update(vocabularyItems)
    .set({ image_url: row.image_url })
    .where(eq(vocabularyItems.id, row.vocab_item_id));
  loaded++;
}
console.log(`[done] loaded=${loaded} skipped=${skipped} invalid=${invalid} total=${tsvRows.length}`);
```

- `import { eq } from "drizzle-orm"` (line 22 of 11-enrich-vocab.ts) — same eq operator on UUID PK.
