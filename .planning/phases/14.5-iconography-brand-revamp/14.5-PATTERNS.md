**ICO binary construction core pattern** (verified in RESEARCH.md — unique to this phase):
```typescript
// Guard: logo.png must be the user's new 512×512 art, not the placeholder
const logoPath = resolve("public/logo.png");
const meta = await sharp(logoPath).metadata();
if (meta.width !== 512 || meta.height !== 512) {
  throw new Error(
    `public/logo.png is ${meta.width}×${meta.height} — expected 512×512. ` +
    `Place the Wave 2 generated logo.png before running this script.`
  );
}

const [buf16, buf32] = await Promise.all([
  sharp(logoPath).resize(16, 16).png().toBuffer(),
  sharp(logoPath).resize(32, 32).png().toBuffer(),
]);

const headerSize = 6;
const dirEntrySize = 16;
const numImages = 2;
const offset16 = headerSize + dirEntrySize * numImages;
const offset32 = offset16 + buf16.length;
const ico = Buffer.alloc(offset32 + buf32.length);

// ICO header (6 bytes): reserved=0, type=1 (icon), count=2
ico.writeUInt16LE(0, 0);
ico.writeUInt16LE(1, 2);
ico.writeUInt16LE(numImages, 4);

// Directory entry 1 — 16×16 (16 bytes at offset 6)
ico.writeUInt8(16, 6);  ico.writeUInt8(16, 7);   // width, height
ico.writeUInt8(0, 8);   ico.writeUInt8(0, 9);    // colorCount, reserved
ico.writeUInt16LE(0, 10); ico.writeUInt16LE(32, 12); // planes, bitCount
ico.writeUInt32LE(buf16.length, 14);              // sizeInBytes
ico.writeUInt32LE(offset16, 18);                  // fileOffset

// Directory entry 2 — 32×32 (16 bytes at offset 22)
ico.writeUInt8(32, 22); ico.writeUInt8(32, 23);
ico.writeUInt8(0, 24);  ico.writeUInt8(0, 25);
ico.writeUInt16LE(0, 26); ico.writeUInt16LE(32, 28);
ico.writeUInt32LE(buf32.length, 30);
ico.writeUInt32LE(offset32, 34);

buf16.copy(ico, offset16);
buf32.copy(ico, offset32);
writeFileSync(resolve("public/favicon.ico"), ico);
console.log(`OK: public/favicon.ico written (${ico.length} bytes)`);
```

**Run command pattern** (from `verse-token-distribution.ts` line 9):

### `scripts/brand/verify-brand-assets.ts` (utility, file-I/O)

**Script header/JSDoc pattern** (from `verse-token-distribution.ts` lines 1–20):
```typescript
/**
 * verify-brand-assets.ts — Validate Wave 2 raster asset dimensions and formats.
 *
 * Checks all 5 public/ brand assets against spec requirements:
 *   - public/logo.png: 512×512, transparent bg allowed
 *   - public/apple-touch-icon.png: 180×180, must be OPAQUE (no alpha)
 *   - public/og-image.png: 1200×630
 *   - public/twitter-image.png: 1200×630
 *   - public/favicon.ico: valid ICO (magic bytes 00000100)
 *
 * Run via: npx tsx --tsconfig tsconfig.scripts.json scripts/brand/verify-brand-assets.ts
 *
 * Exit codes:
 *   0 — all checks pass
 *   1 — one or more checks fail (errors printed to stderr)
 */
```

**Imports pattern** — no dotenv; reads only public/ paths:
```typescript
import sharp from "sharp";
import { readFileSync } from "fs";
import { resolve } from "path";
```

**Spec table + validation loop pattern** (modeled on `verse-token-distribution.ts` constant + loop structure):
```typescript
const EXPECTED = [
  { path: "public/logo.png",             width: 512,  height: 512,  allowAlpha: true  },
  { path: "public/apple-touch-icon.png", width: 180,  height: 180,  allowAlpha: false },
  { path: "public/og-image.png",         width: 1200, height: 630,  allowAlpha: true  },
  { path: "public/twitter-image.png",    width: 1200, height: 630,  allowAlpha: true  },
];

let failures = 0;

for (const spec of EXPECTED) {
  const meta = await sharp(resolve(spec.path)).metadata();
  if (meta.width !== spec.width || meta.height !== spec.height) {
    console.error(`FAIL: ${spec.path}: expected ${spec.width}×${spec.height}, got ${meta.width}×${meta.height}`);
    failures++;
  } else if (!spec.allowAlpha && meta.hasAlpha) {
    console.warn(`WARN: ${spec.path} has alpha channel — apple-touch-icon requires opaque background (iOS renders black square otherwise)`);
    failures++;
  } else {
    console.log(`OK: ${spec.path} — ${meta.width}×${meta.height}${meta.hasAlpha ? " (alpha)" : " (opaque)"}`);
  }
}

// favicon.ico: check magic bytes (ICO = 00 00 01 00)
const icoPath = resolve("public/favicon.ico");
const icoBuf = readFileSync(icoPath);
const icoMagic = icoBuf.readUInt32LE(0);
if (icoMagic !== 0x00010000) {
  console.error(`FAIL: public/favicon.ico — invalid magic bytes (expected 0x00010000, got 0x${icoMagic.toString(16).padStart(8, "0")})`);
  failures++;
} else {
  console.log(`OK: public/favicon.ico — valid ICO magic bytes`);
}

if (failures > 0) {
  console.error(`\n${failures} check(s) failed.`);
  process.exit(1);
}
console.log("\nAll brand asset checks passed.");
```

**Main/exit pattern** — same as synthesize-favicon.ts above.

| Anti-Pattern | Where It Would Appear | Consequence |
|---|---|---|
| `sharp(...).ico().toBuffer()` | synthesize-favicon.ts | TypeError at runtime — sharp 0.34.5 has no `.ico()` method |
