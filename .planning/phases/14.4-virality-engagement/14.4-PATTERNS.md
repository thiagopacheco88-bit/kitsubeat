```typescript
export const getNowPlayingCounts = unstable_cache(
  async (): Promise<Map<string, number>> => {
    const rows = await db.execute(sql`
      SELECT sv.song_id::text, COUNT(DISTINCT COALESCE(sp.user_id, sp.session_key)) AS listener_count
      FROM song_plays sp
      JOIN song_versions sv ON sv.id = sp.song_version_id
      LEFT JOIN users u ON u.id = sp.user_id
      WHERE sp.played_at >= now() - interval '30 minutes'
        AND (sp.user_id IS NULL OR u.social_activity_enabled = true)
      GROUP BY sv.song_id
      HAVING COUNT(DISTINCT COALESCE(sp.user_id, sp.session_key)) >= 3
    `);
    const rawRows = Array.isArray(rows) ? rows : (rows.rows ?? []);
    return new Map(rawRows.map((r: { song_id: string; listener_count: string }) =>
      [r.song_id, Number(r.listener_count)]
    ));
  },
  ["now-playing-counts"],
  { revalidate: 60 }
);
```

**`getActiveOptInUsersForDailyReminder` — plain DB query with timezone filter**:
```typescript
export async function getActiveOptInUsersForDailyReminder(currentHourUtc: number) {
  // Returns users where streak > 0, not logged today in their local tz at 19:00,
  // and their local hour matches currentHourUtc adjusted for their streakTz.
  const rows = await db.execute(sql`
    SELECT id, streak_current, last_streak_date, streak_tz
    FROM users
    WHERE social_activity_enabled = true
      AND streak_current > 0
      AND (last_streak_date IS NULL OR last_streak_date <
        (now() AT TIME ZONE COALESCE(streak_tz, 'UTC'))::date)
      AND EXTRACT(HOUR FROM now() AT TIME ZONE COALESCE(streak_tz, 'UTC')) = 19
  `);
  const rawRows = Array.isArray(rows) ? rows : (rows.rows ?? []);
  return rawRows as Array<{ id: string; streak_current: number; last_streak_date: string | null; streak_tz: string | null }>;
}
```

```typescript
  let skippedNotAtRisk = 0;

  for (const user of users) {

    // Idempotency check — skip if already sent today (D-05 + Pitfall 5)
    const existing = await db.select()
      .from(emailSentLog)
      .where(/* user_id, kind='daily_reminder', period_key=todayKey */)
      .limit(1);
    if (existing.length > 0) { skippedAlreadySent++; continue; }

    // Clerk email lookup (per-send, no mirror column per SPEC constraint #7)
    const clerkUser = await clerkClient().users.getUser(user.id);
    const email = clerkUser.emailAddresses[0]?.emailAddress;
    if (!email) continue;

    const firstName = clerkUser.firstName ?? "Someone";
    const hoursLeft = 19 - new Date().getHours(); // simplified; use tz-aware in impl
    const { subject, html, text } = renderDailyReminder({ firstName, streakCurrent: user.streak_current, hoursLeft });

    // Send — write log AFTER confirm (Pitfall 5)
    const result = await sendEmail({ to: email, subject, html, text, kind: "daily_reminder", userId: user.id });
    if (result.sent || result.dry_run) {
      if (!isDryRun) {
        await db.insert(emailSentLog).values({ user_id: user.id, kind: "daily_reminder", period_key: todayKey });
      }
      emailsSent++;
    }
  }

  return NextResponse.json({
    ok: true,
    kind: "daily_reminder",
    users_evaluated: users.length,
    emails_sent: emailsSent,
    skipped_already_sent: skippedAlreadySent,
    skipped_not_at_risk: skippedNotAtRisk,

}
```

```typescript
  html: string;

  dry_run?: boolean;
  id?: string;
}

export async function sendEmail(payload: SendEmailPayload): Promise<SendEmailResult> {
  if (!process.env.RESEND_API_KEY) {
    // D-05 dry-run: log but do NOT write email_sent_log
    // eslint-disable-next-line no-console

      kind: payload.kind,
      userId: payload.userId,

    to: payload.to,
    subject: payload.subject,
```

### `src/lib/streak/saver.ts` (utility, transform)

**Analog:** `src/lib/gamification/streak.ts` (pure function utility pattern, lines 1–60)

**Pattern** — pure helper functions, no DB imports (DB writes happen in `session-integration.ts`):
```typescript
/**
 * Phase 14.4 D-12 — Streak-saver grant + consume helpers.
 *
 * Pure functions — DB writes happen in applyGamificationUpdate (session-integration.ts).
 * Called from applyGamificationUpdate Step 5.5.
 */

export interface StreakSaverState {
  token: number;  // 0 or 1
  pending: boolean;
}

export interface StreakSaverInput {
  previousStreakCurrent: number;
  streakSaverToken: number;
  milestoneHit: 7 | 30 | 100 | null;
  graceApplied: boolean;
  newStreakCurrent: number;
  todayInTz: string;
}

export interface StreakSaverResult {
  nextToken: number;
  nextPending: boolean;
  nextStreakCurrent: number;
  nextLastStreakDate: string | null;
  consumed: boolean;
  granted: boolean;
}

export function applyStreakSaver(
  input: StreakSaverInput,
  newState: { streakCurrent: number; lastStreakDate: string | null }
): StreakSaverResult {
  let nextToken = input.streakSaverToken;
  let nextPending = false;
  let nextStreakCurrent = newState.streakCurrent;
  let nextLastStreakDate = newState.lastStreakDate;
  let consumed = false;
  let granted = false;

  // Grant rule (D-12): first 7-day milestone, no token held
  if (input.milestoneHit === 7 && nextToken === 0) {
    nextToken = 1;
    granted = true;
  }

  // Consume rule (D-12): reset that grace did not cover, token available
  const wasReset =
    !input.graceApplied &&
    input.previousStreakCurrent > 1 &&
    newState.streakCurrent === 1;

  if (wasReset && nextToken === 1) {
    nextStreakCurrent = input.previousStreakCurrent + 1;
    nextLastStreakDate = input.todayInTz;
    nextToken = 0;
    nextPending = true;
    consumed = true;
    granted = false; // consume wins if both would fire (impossible per D-12 analysis)
  }

  return { nextToken, nextPending, nextStreakCurrent, nextLastStreakDate, consumed, granted };
}
```
