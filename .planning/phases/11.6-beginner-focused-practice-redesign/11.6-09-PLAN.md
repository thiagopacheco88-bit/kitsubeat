    ```tsx
    "use client";

    import { useState, useRef, useEffect } from "react";
    import type { Question } from "@/lib/exercises/generator";
    import { romajiEquals } from "@/lib/exercises/romaji-normalize";
    import KanjiBreakdownSection from "./KanjiBreakdownSection";

    interface Props {
      question: Question;
      onAnswered: (chosen: string, correct: boolean, timeMs: number) => void;
      onContinue: () => void;
      lang?: string;
    }

    export default function VocabTypedCard({ question, onAnswered, onContinue, lang = "en" }: Props) {
      const [input, setInput] = useState("");
      const [submitted, setSubmitted] = useState(false);
      const [isCorrect, setIsCorrect] = useState(false);
      const [imageFailed, setImageFailed] = useState(false);
      const startTimeRef = useRef(Date.now());
      const inputRef = useRef<HTMLInputElement>(null);

      // Reset state when the question changes
      useEffect(() => {
        setInput("");
        setSubmitted(false);
        setIsCorrect(false);
        setImageFailed(false);
        startTimeRef.current = Date.now();
        // Auto-focus the input for fast typing
        inputRef.current?.focus();
      }, [question.id]);

      function handleSubmit(e: React.FormEvent) {
        e.preventDefault();
        if (submitted) return;
        const trimmed = input.trim();
        if (!trimmed) return;
        const correct = romajiEquals(trimmed, question.correctAnswer);
        setIsCorrect(correct);
        setSubmitted(true);
        onAnswered(trimmed, correct, Date.now() - startTimeRef.current);
      }

      const meaningText = question.vocabInfo.meaning_en;
      const hasKanjiBreakdown = !!question.kanji_breakdown;

      return (
        <div className="flex flex-col gap-4 rounded-xl border border-gray-700 bg-gray-900 p-6">
          {/* Image above the prompt (Phase 11.4 pattern) */}
          {question.image_url && !imageFailed && (
            <div className="mb-2 flex justify-center">
              <img
                src={question.image_url}
                alt={meaningText ?? "vocabulary mnemonic"}
                width={224}
                height={224}
                loading="lazy"
                onError={() => setImageFailed(true)}
                data-testid="vocab-typed-image"
                className="aspect-square w-56 h-56 rounded-md object-cover"
              />
            </div>
          )}

          {/* Prompt: kanji surface + meaning hint */}
          <div className="text-center">
            <p className="text-3xl font-semibold text-white">{question.vocabInfo.surface}</p>
            <p className="mt-2 text-sm text-gray-400">{meaningText}</p>
          </div>

          {/* Optional kanji breakdown for Kanji track */}
          {hasKanjiBreakdown && question.kanji_breakdown && (
            <div className="rounded bg-gray-800 p-3">
              <KanjiBreakdownSection breakdown={question.kanji_breakdown} lang={lang} />
            </div>
          )}

          {/* Romaji input */}
          <form onSubmit={handleSubmit} className="flex items-center gap-2">
            <input
              ref={inputRef}
              type="text"
              value={input}
              disabled={submitted}
              placeholder="type the reading in romaji"
              onChange={(e) => setInput(e.target.value)}
              autoFocus
              spellCheck={false}
              autoCapitalize="off"
              autoCorrect="off"
              data-testid="vocab-typed-input"
              className="flex-1 rounded-lg border border-gray-700 bg-gray-800 px-3 py-2 text-base text-white placeholder:text-gray-600 focus:border-red-500 focus:outline-none disabled:opacity-60"
            />
            {!submitted && (
              <button
                type="submit"
                className="rounded-lg bg-red-600 px-4 py-2 text-sm font-medium text-white hover:bg-red-700 disabled:opacity-50"
              >
                Submit
              </button>
            )}
          </form>

          {/* Feedback */}
          {submitted && (
            <div className={`rounded p-3 text-sm ${isCorrect ? "bg-emerald-900 text-emerald-100" : "bg-rose-900 text-rose-100"}`} data-testid="vocab-typed-feedback">
              {isCorrect ? "Correct!" : `Not quite — the answer is "${question.correctAnswer}".`}
              <div className="mt-2 flex justify-end">
                <button
                  type="button"
                  onClick={onContinue}
                  className="rounded bg-white/10 px-3 py-1 text-xs font-medium text-white hover:bg-white/20"
                >
                  Continue
                </button>
              </div>
            </div>
          )}
        </div>
      );
    }
    ```

    NOTE on dangerouslySetInnerHTML: This component does NOT use it for user input. The kanji surface comes from server-generated lesson content (Phase 1) — same threat surface as GrammarWriteCard line 60 per PATTERNS.md. React auto-escaping covers any edge case; no XSS via this surface.

    NOTE on tier.ts: VocabTypedCard does NOT call tierFor. Kanji track is the production surface for typed answers and shows kanji surface + image + meaning directly. SPEC R3.
