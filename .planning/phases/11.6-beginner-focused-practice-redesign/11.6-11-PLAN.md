    ```typescript
    export async function getVocabularyDashboard(userId: string, opts: {
      tierFilter?: 0 | 1 | 2 | 3;
      sourceSongId?: string;
      sortDirection?: "asc" | "desc";
      limit?: number;
    }): Promise<VocabularyDashboardRow[]> {
      const rows = await db.execute<VocabularyDashboardRow>(sql`
        SELECT
          vi.id AS vocab_item_id,
          vi.dictionary_form AS surface,
          vi.reading,
          vi.meaning_en,
          vi.jlpt_level,
          vi.image_url,
          MAX(CASE WHEN m.card_kind = 'romaji_meaning' THEN m.state END) AS romaji_state,
          MAX(CASE WHEN m.card_kind = 'romaji_meaning' THEN m.due END) AS romaji_due,
          MAX(CASE WHEN m.card_kind = 'kanji_kana' THEN m.state END) AS kanji_state,
          MAX(CASE WHEN m.card_kind = 'kanji_kana' THEN m.due END) AS kanji_due
        FROM vocabulary_items vi
        LEFT JOIN user_vocab_mastery m
          ON m.vocab_item_id = vi.id AND m.user_id = ${userId}
        WHERE vi.id IN (
          -- subset to vocab user has encountered (existing Plan 11-04 logic — replicate)
          SELECT DISTINCT m2.vocab_item_id FROM user_vocab_mastery m2 WHERE m2.user_id = ${userId}
        )
        ${opts.sourceSongId ? sql`AND vi.id IN (SELECT vocab_item_id FROM song_vocab WHERE song_version_id = ${opts.sourceSongId}::uuid)` : sql``}
        GROUP BY vi.id, vi.dictionary_form, vi.reading, vi.meaning_en, vi.jlpt_level, vi.image_url
        ${opts.tierFilter !== undefined
          ? sql`HAVING MAX(CASE WHEN m.card_kind = 'romaji_meaning' THEN m.state END) = ${opts.tierFilter}`
          : sql``}
        ORDER BY MAX(CASE WHEN m.card_kind = 'romaji_meaning' THEN m.due END) ${opts.sortDirection === "asc" ? sql`ASC` : sql`DESC`} NULLS LAST
        ${opts.limit ? sql`LIMIT ${opts.limit}` : sql``}
      `);
      return Array.isArray(rows) ? rows : rows.rows ?? [];
    }
    ```
