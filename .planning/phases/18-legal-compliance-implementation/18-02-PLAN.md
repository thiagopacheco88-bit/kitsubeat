    3. Create src/app/actions/consent.ts:
    ```typescript
    "use server";

    import { createHash } from "crypto";
    import { auth } from "@clerk/nextjs/server";
    import { db } from "@/lib/db";
    import { cookieConsentRecord } from "@/lib/db/schema";
    import { CURRENT_COOKIE_CONSENT_VERSION } from "@/lib/legal/versions";
    import { headers } from "next/headers";

    export async function recordConsent(decision: "granted" | "rejected"): Promise<void> {
      const { userId } = await auth();
      const headerStore = await headers();
      const ip = headerStore.get("x-forwarded-for")?.split(",")[0]?.trim() ?? "";
      const userAgent = headerStore.get("user-agent") ?? undefined;

      // REQ-PRIV-COOKIE-05: store SHA-256 hash of IP, NEVER raw IP (GDPR Art. 5(1)(c))
      const ipHash = ip
        ? createHash("sha256").update(ip + (process.env.CRON_SECRET ?? "kitsubeat-salt")).digest("hex")
        : undefined;

      await db.insert(cookieConsentRecord).values({
        user_id: userId ?? null,
        consent_version: CURRENT_COOKIE_CONSENT_VERSION,
        categories: { essential: true, analytics: decision === "granted" },
        decision,
        ip_hash: ipHash,
        user_agent: userAgent,
      });

      // Set browser cookie via dynamic import (prevents jsdom crash in unit tests)
      const { cookies } = await import("next/headers");
      const c = await cookies();
      c.set("kb_consent", decision, {
        maxAge: 60 * 60 * 24 * 365,
        sameSite: "lax",
        httpOnly: false,
        path: "/",
      });
    }
    ```

    ```typescript
    "use server";

    import { auth, clerkClient } from "@clerk/nextjs/server";
    import { db } from "@/lib/db";
    import { users } from "@/lib/db/schema";
    import { CURRENT_TERMS_VERSION } from "@/lib/legal/versions";

    export async function completeOnboarding(dateOfBirth: string): Promise<{ error?: string }> {
      const { userId } = await auth();
      if (!userId) return { error: "Not authenticated" };

      // Server re-validates DOB regardless of client (REQ-MINORS-GATE-06)
      const dob = new Date(dateOfBirth);
      if (isNaN(dob.getTime())) return { error: "invalid_date" };
      const now = new Date();
      const ageYears = (now.getTime() - dob.getTime()) / (1000 * 60 * 60 * 24 * 365.25);

      if (ageYears < 13) return { error: "under_13" }; // REQ-MINORS-GATE-03

      const isMinor = ageYears < 18;

      // CRITICAL (Pitfall 7): all defaults applied atomically in ONE upsert — never two DB calls
      await db
        .insert(users)
        .values({
          id: userId,
          date_of_birth: dateOfBirth,
          is_minor: isMinor,
          terms_accepted_at: now,
          terms_version: CURRENT_TERMS_VERSION,
          minor_defaults_applied: isMinor,
          ...(isMinor ? {
            social_activity_enabled: false,  // REQ-MINORS-DEFAULT-01 / REQ-MINORS-12
            marketing_email_opt_in: false,   // REQ-MINORS-DEFAULT-03
          } : {}),
        })
        .onConflictDoUpdate({
          target: users.id,
          set: {
            date_of_birth: dateOfBirth,
            is_minor: isMinor,
            terms_accepted_at: now,
            terms_version: CURRENT_TERMS_VERSION,
            minor_defaults_applied: isMinor,
            ...(isMinor ? {
              social_activity_enabled: false,
              marketing_email_opt_in: false,
            } : {}),
          },
        });

      // Cache terms_version in Clerk publicMetadata (0ms DB query in middleware — RESEARCH Open Question 2)
      const client = await clerkClient();
      await client.users.updateUserMetadata(userId, {
        publicMetadata: { terms_version: CURRENT_TERMS_VERSION },
      });

      return {};
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → completeOnboarding() server action | User-submitted DOB; must be server-re-validated |
| Client → recordConsent() server action | Unauthenticated callers allowed; consent decision must match user action |
| Authenticated request → /api/user/data-export | Must return only the authenticated user's data — never another user's |
| Cron caller → /api/cron/birthday-transitions | Must validate CRON_SECRET header |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-18-02-01 | Tampering | completeOnboarding() DOB input | mitigate | Server re-validates DOB regardless of client state (REQ-MINORS-GATE-06); ageYears computed server-side from date arithmetic |
| T-18-02-02 | Information Disclosure | /api/user/data-export | mitigate | Filter all 5 table queries by `auth().userId` — never by a client-supplied userId param; integration test in Wave 0 tests two-user isolation |
| T-18-02-03 | Tampering | cookie_consent_record IP field | mitigate | Store SHA-256 hash via Node crypto.createHash — never raw IP (GDPR Art. 5(1)(c) data minimization) |
| T-18-02-04 | Spoofing | recordConsent() called without user action | mitigate | Server action is Clerk-authenticated (userId available for audit); anonymous consent tracked by session/IP hash only; no elevation of consent state without server action call |
| T-18-02-05 | Spoofing | /api/cron/birthday-transitions unauthorized call | mitigate | assertCronSecret validates Bearer token; unauthenticated requests return 401 before any DB mutation |
| T-18-02-06 | Tampering | birthday cron changes extra settings | mitigate | db.update().set({ is_minor: false }) — explicitly only this one column; Pitfall 5 documented; unit test will verify no other column modified |
</threat_model>
