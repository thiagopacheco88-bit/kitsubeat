```tsx
/**
 * TierDivider — Phase 14.1 SPEC-REQ-7.
 *
 * Bilingual co-headed tier marker on /path. Replaces the inline tier-divider
 * <span> at PathMap.tsx:58-71. Renders a 1px-rule + chip layout with a
 * tier-specific SVG icon and a Japanese·English co-heading.
 *
 * Tier labels (CONTEXT Specifics):
 *   basic        -> Side A · 基礎 · Beginner    (bamboo icon)
 *   intermediate -> Side B · 中級 · Intermediate (torii icon)
 *   advanced     -> Side C · 上級 · Advanced     (mountain icon)
 *
 * Icons are hand-authored 16×16 SVGs per Claude's Discretion; Lucide doesn't
 * provide torii or bamboo. All fills are `currentColor` so the icon inherits
 * the chip's text color.
 *
 * Server component — no client interactivity.
 */

type Tier = "basic" | "intermediate" | "advanced";

interface TierDividerProps {
  tier: Tier | string;
}

interface TierMeta {
  sideLabel: string;        // "Side A"
  jpLabel: string;          // "基礎"
  enLabel: string;          // "Beginner"
  iconKind: "bamboo" | "torii" | "mountain";
}

const TIER_META: Record<Tier, TierMeta> = {
  basic: {
    sideLabel: "Side A",
    jpLabel: "基礎",
    enLabel: "Beginner",
    iconKind: "bamboo",
  },
  intermediate: {
    sideLabel: "Side B",
    jpLabel: "中級",
    enLabel: "Intermediate",
    iconKind: "torii",
  },
  advanced: {
    sideLabel: "Side C",
    jpLabel: "上級",
    enLabel: "Advanced",
    iconKind: "mountain",
  },
};

function TierIcon({ kind }: { kind: TierMeta["iconKind"] }) {
  switch (kind) {
    case "bamboo":
      return (
        <svg
          width="16"
          height="16"
          viewBox="0 0 16 16"
          fill="currentColor"
          aria-hidden="true"
          data-testid="tier-icon-bamboo"
        >
          {/* Hand-authored bamboo: vertical stalk with 3 horizontal nodes
           * + 2 leaf flares. Executor: refine against demo if present. */}
          <rect x="7" y="1" width="2" height="14" rx="0.5" />
          <rect x="6" y="4" width="4" height="0.6" />
          <rect x="6" y="8" width="4" height="0.6" />
          <rect x="6" y="12" width="4" height="0.6" />
          <path d="M9 5 Q12 4 13 6 Q11 6 9 5 Z" />
          <path d="M7 9 Q4 8 3 10 Q5 10 7 9 Z" />
        </svg>
      );
    case "torii":
      return (
        <svg
          width="16"
          height="16"
          viewBox="0 0 16 16"
          fill="currentColor"
          aria-hidden="true"
          data-testid="tier-icon-torii"
        >
          {/* Hand-authored torii gate: top crossbeam (kasagi) + lower
           * crossbeam (nuki) + 2 vertical pillars (hashira). */}
          <rect x="1" y="2" width="14" height="2" />
          <rect x="2" y="5" width="12" height="1.4" />
          <rect x="3" y="2" width="2" height="13" />
          <rect x="11" y="2" width="2" height="13" />
        </svg>
      );
    case "mountain":
      return (
        <svg
          width="16"
          height="16"
          viewBox="0 0 16 16"
          fill="currentColor"
          aria-hidden="true"
          data-testid="tier-icon-mountain"
        >
          {/* Hand-authored mountain: triangle peaks with snow cap.
           * Three peaks for "advanced" gravitas. */}
          <path d="M1 14 L5 6 L8 11 L11 5 L15 14 Z" />
          <path d="M4 8 L5 6 L6 8 Z" fill="var(--color-bg)" />
          <path d="M10 7 L11 5 L12 7 Z" fill="var(--color-bg)" />
        </svg>
      );
  }
}

export function TierDivider({ tier }: TierDividerProps) {
  const meta = TIER_META[tier as Tier];

  // Defensive fallback: unknown tier -> render input as label, no icon
  if (!meta) {
    return (
      <div
        className="flex items-center gap-3 my-4"
        aria-label={`Tier: ${tier}`}
      >
        <div className="flex-1 border-t border-[var(--color-border)]" />
        <span className="rounded-[var(--radius-pill)] border border-[var(--color-border-strong)] bg-[var(--color-card-2)] px-3 py-0.5 text-xs font-semibold text-[var(--color-text-muted)] uppercase tracking-wide">
          {tier}
        </span>
        <div className="flex-1 border-t border-[var(--color-border)]" />
      </div>
    );
  }

  const fullLabel = `${meta.sideLabel} · ${meta.jpLabel} · ${meta.enLabel}`;

  return (
    <div
      className="flex items-center gap-3 my-4"
      aria-label={fullLabel}
      data-testid={`tier-divider-${tier}`}
    >
      <div className="flex-1 border-t border-[var(--color-border)]" />
      <span className="inline-flex items-center gap-1.5 rounded-[var(--radius-pill)] border border-[var(--color-border-strong)] bg-[var(--color-card-2)] px-3 py-0.5 text-xs font-semibold text-[var(--color-text-muted)] uppercase tracking-wide">
        <TierIcon kind={meta.iconKind} />
        <span>{meta.sideLabel}</span>
        <span style={{ fontFamily: "var(--font-jp)" }} className="normal-case">
          {meta.jpLabel}
        </span>
        <span>{meta.enLabel}</span>
      </span>
      <div className="flex-1 border-t border-[var(--color-border)]" />
    </div>
  );
}
```

Implementation rules:
1. **Verify SVG icons against demo.** Open `_temp/path-redesign/demo-CA-hybrid.html` and find the TierDivider section. If the demo provides specific SVG path data for bamboo/torii/mountain, COPY THE DEMO'S PATHS verbatim — the placeholders above are starting sketches the executor refines against the visual contract.
2. **Icon fills are `currentColor`** (or `var(--color-bg)` for snow-cap inversion details) — never raw hex. The chip's `text-[var(--color-text-muted)]` cascades to the SVG via `fill="currentColor"`.
3. **`<span style={{ fontFamily: "var(--font-jp)" }}>`** for the Japanese co-heading. This + `normal-case` is needed because the parent chip applies `uppercase tracking-wide` which mangles Japanese characters.
4. **Bilingual `aria-label`** on the OUTER `<div>` carries the full `Side A · 基礎 · Beginner` string for screen readers.
5. **`data-testid="tier-divider-{tier}"`** on the outer div for E2E selection.
6. **`data-testid="tier-icon-{kind}"`** on each SVG for the per-tier icon test.
7. **Defensive fallback:** if `tier` is not one of the 3 known values, render the input string in a chip with no icon and don't crash. Old PathMap had this graceful degradation; preserve it.
