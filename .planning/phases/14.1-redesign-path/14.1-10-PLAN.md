```tsx
"use client";

/**
 * ContinueAnchor — Phase 14.1 SPEC-REQ-8.
 *
 * Sticky-bottom CTA on /path. Renders only when the user has a current song
 * (current_path_node_slug !== null). On the StarterPick branch (null slug),
 * this component is NOT mounted — page.tsx handles the conditional in Plan 11.
 *
 * Layout:
 *   position: fixed; bottom: 0; full-width with horizontal padding (matches
 *   /path's mx-auto max-w-2xl px-4 column on the parent).
 *   min-height: 56px (≥ Phase 14 SPEC AC #11 44px tap-target floor).
 *   Background: var(--color-accent) (brand red); glow: var(--shadow-cta-red)
 *   (existing token from Phase 14-01).
 *
 * Above the CTA, a gradient-fade <div> bleeds the page bg color upward so
 * scrolling content reads as continuing underneath the CTA, not abruptly cut.
 *
 * On tap: navigate to /songs/{currentSongSlug}.
 */
import Link from "next/link";

interface ContinueAnchorProps {
  currentSongSlug: string | null;
  currentSongTitle: string | null;
}

export function ContinueAnchor({
  currentSongSlug,
  currentSongTitle,
}: ContinueAnchorProps) {
  // Visibility guard: render nothing if either piece is missing
  if (!currentSongSlug || !currentSongTitle) return null;

  return (
    <>
      {/* Gradient-fade above the CTA so scrolling content blends in */}
      <div
        className="fixed bottom-[56px] left-0 right-0 h-12 pointer-events-none z-40"
        style={{
          background:
            "linear-gradient(180deg, transparent 0%, var(--color-bg) 100%)",
        }}
        aria-hidden="true"
      />
      {/* Sticky CTA itself */}
      <div className="fixed bottom-0 left-0 right-0 z-50 px-4 py-3 bg-[var(--color-bg)]">
        <Link
          href={`/songs/${currentSongSlug}`}
          className="flex items-center justify-center gap-2 w-full min-h-[56px] rounded-[var(--radius-lg)] bg-[var(--color-accent)] [color:white] font-semibold text-base shadow-[var(--shadow-cta-red)] hover:opacity-95 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--color-accent)]/40 transition-opacity"
          data-testid="continue-anchor"
          aria-label={`Continue: ${currentSongTitle}`}
        >
          <span>Continue</span>
          <span aria-hidden="true">·</span>
          <span
            className="truncate max-w-[60%]"
            style={{ fontFamily: "var(--font-jp)" }}
          >
            {currentSongTitle}
          </span>
        </Link>
      </div>
    </>
  );
}
```

1. **Verify against demo.** Open `_temp/path-redesign/demo-CA-hybrid.html` and find the ContinueAnchor / sticky CTA. Match label format ("Continue · {title}" vs another), gradient-fade height + color stops, padding around CTA. The placeholder above is a starting structure; refine to the demo.
2. **`"use client"`** is needed because `<Link>` from next/link requires the client runtime for prefetching + soft navigation (although it can server-render, the hover/click handlers are client). Adding the directive is the safe choice.
3. **`fixed bottom-0`** Tailwind utilities are required — the M2 fixed-bottom assertion in Plan 12 Playwright spec checks for `position: fixed` via computed style.
4. **`min-h-[56px]`** as a Tailwind arbitrary value (NOT inline `style`). E2E test checks the className includes `min-h-[56px]`.
5. **Gradient-fade element** is a SEPARATE `<div>` ABOVE the CTA wrapper. It uses `bottom-[56px]` to sit on top of the CTA, has `pointer-events-none` so clicks pass through. Background gradient: transparent at top, `var(--color-bg)` at bottom — content fades into the page bg as it approaches the CTA.
6. **`shadow-[var(--shadow-cta-red)]`** — the existing token from Phase 14-01 (no need to author; CONTEXT D-13 confirmed during scout).
7. **`[color:white]`** Tailwind v4 arbitrary-property syntax for white text on accent background — matches the kana page CTA pattern + dodges any `bareWhiteBlack` audit (path/components is NOT in the lint allowlist; arbitrary-property syntax is the established codebase escape).
8. **z-index**: gradient at `z-40`, CTA at `z-50`. Anything below stays under both.
9. **`px-4 py-3`** on the CTA wrapper provides side padding so the rounded button doesn't bleed to viewport edges; matches the parent /path's `px-4` column padding.
10. **`data-testid="continue-anchor"`** on the `<Link>` itself — Playwright Plan 12 selects by this hook.
11. **`aria-label`** on the link gives screen readers the full meaning. The visible text is "Continue · {title}" which is also accessible.

import { describe, it, expect } from "vitest";
import { render } from "@testing-library/react";
import { ContinueAnchor } from "./ContinueAnchor";

describe("ContinueAnchor", () => {
  it("renders nothing when currentSongSlug is null", () => {
    const { container } = render(
      <ContinueAnchor currentSongSlug={null} currentSongTitle="X" />,
    );
    expect(container.innerHTML).toBe("");
  });

  it("renders nothing when currentSongTitle is null", () => {
    const { container } = render(
      <ContinueAnchor currentSongSlug="slug" currentSongTitle={null} />,
    );
    expect(container.innerHTML).toBe("");
  });

  it("renders nothing when currentSongTitle is empty string", () => {
    const { container } = render(
      <ContinueAnchor currentSongSlug="slug" currentSongTitle="" />,
    );
    expect(container.innerHTML).toBe("");
  });

  it("renders with href '/songs/{slug}' when both props valid", () => {
    const { getByTestId } = render(
      <ContinueAnchor
        currentSongSlug="silhouette-kanaboon"
        currentSongTitle="シルエット"
      />,
    );
    const link = getByTestId("continue-anchor");
    expect(link.getAttribute("href")).toBe("/songs/silhouette-kanaboon");
  });

  it("visible text contains 'Continue' and the song title", () => {
    const { container } = render(
      <ContinueAnchor
        currentSongSlug="silhouette-kanaboon"
        currentSongTitle="シルエット"
      />,
    );
    expect(container.textContent).toContain("Continue");
    expect(container.textContent).toContain("シルエット");
  });

  it("CTA element has fixed-bottom positioning via Tailwind utilities", () => {
    const { container, getByTestId } = render(
      <ContinueAnchor
        currentSongSlug="slug"
        currentSongTitle="title"
      />,
    );
    // The CTA's parent wrapper carries the fixed-bottom-0 utilities.
    const link = getByTestId("continue-anchor");
    const wrapper = link.parentElement;
    expect(wrapper?.className).toContain("fixed");
    expect(wrapper?.className).toContain("bottom-0");
  });

  it("CTA carries min-h-[56px] class for tap-target compliance", () => {
    const { getByTestId } = render(
      <ContinueAnchor currentSongSlug="x" currentSongTitle="y" />,
    );
    expect(getByTestId("continue-anchor").className).toContain("min-h-[56px]");
  });

  it("aria-label on the link includes 'Continue:' and the title", () => {
    const { getByTestId } = render(
      <ContinueAnchor currentSongSlug="x" currentSongTitle="シルエット" />,
    );
    expect(getByTestId("continue-anchor").getAttribute("aria-label")).toBe(
      "Continue: シルエット",
    );
  });
});
