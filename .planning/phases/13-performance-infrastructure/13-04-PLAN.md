Output: Two npm scripts (`lighthouse:pick-target`, `lighthouse:baseline`), a target-song.txt artifact, six raw Lighthouse JSON outputs, and a Performance baseline section in 13-SUMMARY.md.

    **(a) scripts/lighthouse-pick-target.ts** — DB-read + median pick + file write:

    Create file with:
    ```typescript
    /**
     * Phase 13 D-17: pick the catalog-median song by lesson JSONB byte size.
     *
     * Writes the chosen slug to lighthouse-baseline/target-song.txt so subsequent
     * `npm run lighthouse:baseline` runs (and Phase 19 re-runs) target the SAME
     * representative slug.
     *
     * Pre-condition: .env.local has DATABASE_URL set. The dotenv load MUST happen
     * before any DB-touching import (the db Proxy validates DATABASE_URL on first
     * access — see src/lib/db/index.ts:91-100).
     *
     * Usage:
     *   npm run lighthouse:pick-target
     */
    import { config } from "dotenv";
    import { resolve, join } from "path";
    import { fileURLToPath } from "url";
    import { writeFileSync, mkdirSync } from "fs";

    const __dirname = fileURLToPath(new URL(".", import.meta.url));
    config({ path: join(resolve(__dirname, ".."), ".env.local") });

    // Lazy DB import — must come AFTER dotenv.
    const { db } = await import("../src/lib/db/index.js");
    const { sql } = await import("drizzle-orm");

    const rows = (await db.execute(sql`
      SELECT s.slug, octet_length(v.lesson::text) AS bytes
        FROM songs s
        JOIN song_versions v ON v.song_id = s.id
       WHERE v.lesson IS NOT NULL
       ORDER BY bytes ASC
    `)) as unknown as Array<{ slug: string; bytes: number }>;

    if (rows.length === 0) {
      console.error("[pick-target] No seeded songs with lesson found. Run npm run seed:dev first.");
      process.exit(1);
    }

    const median = rows[Math.floor(rows.length / 2)];
    const outDir = resolve(__dirname, "..", ".planning", "phases", "13-performance-infrastructure", "lighthouse-baseline");
    mkdirSync(outDir, { recursive: true });
    writeFileSync(resolve(outDir, "target-song.txt"), median.slug, "utf-8");

    console.log(
      `[pick-target] median slug: ${median.slug} (${median.bytes} bytes; rank ${Math.floor(rows.length / 2) + 1}/${rows.length})`
    );

    process.exit(0);
    ```

    Note: The result of `db.execute(...)` may be either an array OR an object with `.rows` depending on the Drizzle/neon-http driver. Use the defensive pattern from PATTERNS.md if needed:
    ```typescript
    const result = await db.execute(...);
    const rows = (Array.isArray(result) ? result : (result as { rows?: unknown }).rows ?? []) as Array<{ slug: string; bytes: number }>;
    ```

    **(b) scripts/lighthouse-baseline.ts** — Lighthouse orchestration:

    Create file with:
    ```typescript
    /**
     * Phase 13 D-14/D-15/D-16: capture local Lighthouse baseline.
     *
     * Pre-condition: `npm run start` (production build) MUST be running in a separate
     * shell on http://localhost:7000. Lighthouse is run against the production build,
     * NOT the dev server, per D-15 (matches what Phase 19 entry gate will run).
     *
     * Routes: / (home), /songs (catalog), /songs/{slug from target-song.txt} (song page).
     * Presets: mobile (Phase 19 gate target — Moto G4 4G profile) + desktop (ceiling).
     *
     * Usage:
     *   # In shell A:
     *   npm run build && npm run start    # production build on :7000
     *
     *   # In shell B:
     *   npm run lighthouse:pick-target    # populate target-song.txt
     *   npm run lighthouse:baseline       # ~3-5 min for 6 runs
     *
     * Output: .planning/phases/13-performance-infrastructure/lighthouse-baseline/{home,catalog,song}-{mobile,desktop}.json
     */
    import { spawnSync } from "node:child_process";
    import { existsSync, mkdirSync, readFileSync } from "node:fs";
    import { resolve } from "node:path";
    import { fileURLToPath } from "node:url";

    const __dirname = fileURLToPath(new URL(".", import.meta.url));
    const OUT_DIR = resolve(
      __dirname,
      "..",
      ".planning",
      "phases",
      "13-performance-infrastructure",
      "lighthouse-baseline"
    );
    const BASE = "http://localhost:7000";

    mkdirSync(OUT_DIR, { recursive: true });

    const targetFile = resolve(OUT_DIR, "target-song.txt");
    if (!existsSync(targetFile)) {
      console.error(
        "[baseline] target-song.txt not found. Run `npm run lighthouse:pick-target` first."
      );
      process.exit(1);
    }
    const targetSlug = readFileSync(targetFile, "utf-8").trim();
    if (!targetSlug) {
      console.error("[baseline] target-song.txt is empty.");
      process.exit(1);
    }

    interface Run {
      label: string;
      route: string;
      preset: "mobile" | "desktop";
      outFile: string;
    }

    const RUNS: Run[] = [
      { label: "home", route: "/", preset: "mobile", outFile: "home-mobile.json" },
      { label: "home", route: "/", preset: "desktop", outFile: "home-desktop.json" },
      { label: "catalog", route: "/songs", preset: "mobile", outFile: "catalog-mobile.json" },
      { label: "catalog", route: "/songs", preset: "desktop", outFile: "catalog-desktop.json" },
      { label: "song", route: `/songs/${targetSlug}`, preset: "mobile", outFile: "song-mobile.json" },
      { label: "song", route: `/songs/${targetSlug}`, preset: "desktop", outFile: "song-desktop.json" },
    ];

    let failed = 0;
    for (const r of RUNS) {
      const url = `${BASE}${r.route}`;
      const out = resolve(OUT_DIR, r.outFile);
      console.log(`[baseline] ${r.label}-${r.preset}: ${url} → ${r.outFile}`);
      const args = [
        "lighthouse",
        url,
        `--preset=${r.preset === "mobile" ? "mobile" : "desktop"}`,
        "--output=json",
        `--output-path=${out}`,
        "--chrome-flags=--headless --no-sandbox",
        "--quiet",
      ];
      const proc = spawnSync("npx", args, { stdio: "inherit", shell: process.platform === "win32" });
      if (proc.status !== 0) {
        console.error(`[baseline] FAILED: ${r.label}-${r.preset} exit=${proc.status}`);
        failed++;
      }
    }

    if (failed > 0) {
      console.error(`[baseline] ${failed} of ${RUNS.length} runs failed.`);
      process.exit(1);
    }

    console.log(`[baseline] All ${RUNS.length} Lighthouse runs complete. Output: ${OUT_DIR}`);
    console.log(
      `[baseline] Next: extract scores into 13-SUMMARY.md "Performance baseline" table (Plan 13-04 Task 2).`
    );
    process.exit(0);
    ```
