    ```typescript
    /**
     * HeroFeatured - Phase 14.2 SPEC §Req 3 + AC #6/7/8/9.
     *
     * 280px full-bleed cover-art hero block at the top of /. Server component; consumes
     * HeroSongResult from Plan 14.2-03 getHeroSong(userId | null) which encapsulates
     * auth-aware dispatch + fallback (CONTEXT D-04). page.tsx becomes thin:
     *   const hero = await getHeroSong(userId);
     *   <HeroFeatured hero={hero} />
     *
     * CTA label varies by source (CONTEXT D-04):
     *   current_path       -> "Resume Lesson"
     *   unauth_featured    -> "Try Free Lesson"
     *   fallback_featured  -> "Start Learning"
     *
     * Token discipline (CONTEXT D-12): reuses --shadow-cta-red on CTA, --shadow-hero-glow
     * on container. No new tokens. Cover bleed via maxresdefault.jpg with mqdefault.jpg
     * onError fallback (per SPEC §Req 3).
     *
     * M1 invariant (CONTEXT D-13): CTA anchor clickable; vignette overlay carries
     * pointer-events: none. Tap target ≥56px (CONTEXT D-15).
     */
    import Link from "next/link";
    import type { HeroSongResult } from "@/lib/db/queries";

    interface HeroFeaturedProps {
      hero: HeroSongResult;
    }

    export function HeroFeatured({ hero }: HeroFeaturedProps) {
      const { song, ctaLabel, ctaHref } = hero;
      const coverSrc = song.youtube_id
        ? `https://img.youtube.com/vi/${song.youtube_id}/maxresdefault.jpg`
        : null;
      const coverFallback = song.youtube_id
        ? `https://img.youtube.com/vi/${song.youtube_id}/mqdefault.jpg`
        : null;

      const metaParts = [song.artist, song.anime];
      if (song.verse_count > 0) {
        metaParts.push(`${song.verse_count} verses`);
      }
      const metaLine = metaParts.join(" · ");

      return (
        <section
          data-testid="hero-featured"
          className="relative h-[280px] overflow-hidden rounded-[var(--radius-xl)] mb-6 shadow-[var(--shadow-hero-glow)]"
          aria-label="Featured song"
        >
          {/* Cover - full bleed */}
          {coverSrc ? (
            <img
              src={coverSrc}
              alt={song.title}
              className="absolute inset-0 w-full h-full object-cover"
              data-testid="hero-cover"
              onError={(e) => {
                const img = e.currentTarget;
                if (coverFallback && img.src !== coverFallback) img.src = coverFallback;
              }}
            />
          ) : (
            <div
              className="absolute inset-0 flex items-center justify-center bg-[var(--color-card-2)] text-5xl font-bold text-[var(--color-text-dim)]"
              aria-hidden="true"
              data-testid="hero-placeholder"
            >
              ♪
            </div>
          )}

          {/* Bottom vignette - pointer-events: none (M1) */}
          <div
            className="absolute inset-x-0 bottom-0 h-3/4"
            style={{
              background:
                "linear-gradient(180deg, transparent 0%, rgba(0,0,0,0.85) 100%)",
              pointerEvents: "none",
            }}
            aria-hidden="true"
          />

          {/* Eyebrow row: Featured this week + JLPT chip */}
          <div className="absolute top-3 left-3 right-3 flex items-center justify-between gap-2">
            <span
              className="rounded-[var(--radius-pill)] bg-white/15 backdrop-blur-sm px-3 py-1 text-xs font-bold text-white"
              data-testid="hero-eyebrow"
            >
              ★ Featured this week
            </span>
            {song.jlpt_level && (
              <span
                className="rounded-[var(--radius-pill)] bg-[var(--color-accent)] px-2 py-1 text-xs font-bold text-white"
                data-testid="hero-jlpt-chip"
              >
                JLPT {song.jlpt_level}
              </span>
            )}
          </div>

          {/* Bottom stack: title + meta + CTA */}
          <div className="absolute inset-x-0 bottom-0 p-4 flex flex-col gap-3">
            <h1
              className="font-black text-white truncate"
              style={{
                fontFamily: "var(--font-jp)",
                fontSize: "28px",
                fontWeight: 900,
                textShadow: "0 2px 4px rgba(0,0,0,0.65)",
              }}
              data-testid="hero-title-jp"
            >
              {song.title}
            </h1>
            <p
              className="truncate text-sm text-white/90"
              style={{ textShadow: "0 1px 2px rgba(0,0,0,0.55)" }}
              data-testid="hero-meta"
            >
              {metaLine}
            </p>
            <Link
              href={ctaHref}
              className="inline-flex h-14 min-h-[56px] items-center justify-center gap-2 self-start rounded-[var(--radius-pill)] bg-white px-6 text-sm font-bold text-[var(--color-accent)] shadow-[var(--shadow-cta-red)] transition-transform hover:scale-105"
              data-testid="hero-cta"
            >
              <span aria-hidden="true">▶</span>
              {ctaLabel}
            </Link>
          </div>
        </section>
      );
    }
    ```
