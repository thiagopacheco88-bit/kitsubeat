    ```typescript
    import AxeBuilder from "@axe-core/playwright";
    // Per revision: authenticated axe runs use authenticatedTest (Plan 14.2-01b bypass).
    // Anonymous axe runs use the regular @playwright/test base (the bypass is opt-in).
    import { authenticatedTest, expect } from "../support/auth-fixtures";

    /**
     * SPEC AC #5 — axe-core a11y scan of / >= 95 in 4 configurations:
     *   (auth, dark) (auth, light) (anon, dark) (anon, light)
     *
     * Zero `serious` or `critical` violations per CONTEXT D-11 (carry from 14.1).
     */
    interface Config {
      name: string;
      theme: "dark" | "light";
      authenticated: boolean;
    }

    const CONFIGS: Config[] = [
      { name: "auth + dark", theme: "dark", authenticated: true },
      { name: "auth + light", theme: "light", authenticated: true },
      { name: "anon + dark", theme: "dark", authenticated: false },
      { name: "anon + light", theme: "light", authenticated: false },
    ];

    /**
     * Per revision Edit 4 (option a): split into TWO describe blocks, one per
     * fixture. The bare `test` symbol is undefined at module scope (we only
     * import `authenticatedTest` and `anonTest`), so a single `test.describe`
     * would fail to compile / fail at runtime with "test is not defined". Each
     * fixture exposes its own `.describe`, so we use `authenticatedTest.describe`
     * for the 2 auth configs and `anonTest.describe` for the 2 anon configs.
     */

    // Shared scan body — extracted so both describe blocks call the same logic.
    async function runAxeScan(
      page: import("@playwright/test").Page,
      context: import("@playwright/test").BrowserContext,
      config: Config,
    ) {
      await page.setViewportSize({ width: 390, height: 844 });
      await context.addCookies([
        { name: "kb_theme", value: config.theme, url: "http://localhost:7000", sameSite: "Lax" },
      ]);

      await page.goto("/");
      await page.waitForLoadState("networkidle");

      const result = await new AxeBuilder({ page })
        .withTags(["wcag2a", "wcag2aa", "wcag21a", "wcag21aa"])
        .analyze();

      // Filter to serious + critical only (SPEC AC #5)
      const blocking = result.violations.filter(
        (v) => v.impact === "serious" || v.impact === "critical",
      );

      if (blocking.length > 0) {
        console.log(
          `[axe ${config.name}] BLOCKING violations:`,
          JSON.stringify(
            blocking.map((v) => ({
              id: v.id,
              impact: v.impact,
              description: v.description,
              nodes: v.nodes.length,
            })),
            null,
            2,
          ),
        );
      }

      expect(blocking).toHaveLength(0);
    }

    authenticatedTest.describe("/ axe-core a11y >= 95 (auth contexts) (AC #5)", () => {
      for (const config of CONFIGS.filter((c) => c.authenticated)) {
        authenticatedTest(`${config.name} — zero serious|critical violations`, async ({ page, context }) => {
          await runAxeScan(page, context, config);
        });
      }
    });

    anonTest.describe("/ axe-core a11y >= 95 (anon contexts) (AC #5)", () => {
      for (const config of CONFIGS.filter((c) => !c.authenticated)) {
        anonTest(`${config.name} — zero serious|critical violations`, async ({ page, context }) => {
          await runAxeScan(page, context, config);
        });
      }
    });
    ```
