    Step 2 — Create `src/lib/journal/articles.ts`:

    ```typescript
    import 'server-only';
    import fs from 'node:fs';
    import path from 'node:path';
    import matter from 'gray-matter';
    import readingTime from 'reading-time';

    const CONTENT_DIR = path.join(process.cwd(), 'src/content/journal');

    export type ArticleCategory = 'lore' | 'language' | 'translation';

    export interface ArticleFrontmatter {
      title: string;
      slug: string;
      date: string;
      coverImage: string;
      category: ArticleCategory;
      summary: string;
      tags?: string[];
      author?: string;
      readingTime?: string;
    }

    export interface ArticleMeta extends ArticleFrontmatter {
      readingTimeComputed: string;
    }

    export function getAllArticles(): ArticleMeta[] {
      if (!fs.existsSync(CONTENT_DIR)) return [];
      const files = fs.readdirSync(CONTENT_DIR).filter((f) => f.endsWith('.mdx'));
      return files
        .map((filename) => {
          const raw = fs.readFileSync(path.join(CONTENT_DIR, filename), 'utf8');
          const { data, content } = matter(raw);
          const rt = readingTime(content);
          return {
            ...(data as ArticleFrontmatter),
            readingTimeComputed: (data as ArticleFrontmatter).readingTime ?? rt.text,
          };
        })
        .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
    }

    // Slug guard: prevents path traversal. Only lowercase letters, digits, and hyphens.
    const SAFE_SLUG_RE = /^[a-z0-9-]+$/;

    export function getArticleSource(slug: string): string | null {
      if (!SAFE_SLUG_RE.test(slug)) return null;
      const filepath = path.join(CONTENT_DIR, `${slug}.mdx`);
      if (!fs.existsSync(filepath)) return null;
      return fs.readFileSync(filepath, 'utf8');
    }
    ```

    IMPORTANT: `import 'server-only'` is the first import. This gives a hard build error if `articles.ts` is accidentally imported in a Client Component (per Common Pitfall #4 in RESEARCH.md). The `server-only` package ships with Next.js — no npm install needed.

    ```typescript
    // src/lib/journal/__tests__/articles.test.ts
    // @vitest-environment node
    import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';

    // Mock fs BEFORE importing the module under test so the module picks up mocks
    vi.mock('node:fs');
    import * as fs from 'node:fs';
    import { getAllArticles, getArticleSource } from '../articles';

    const FIXTURE_A = `---
    title: "Article A"
    slug: "article-a"
    date: "2026-05-10"
    coverImage: "/images/journal/article-a.jpg"
    category: "lore"
    summary: "Summary of article A"
    ---
    This is the body of article A with some words for reading time.`;

    const FIXTURE_B = `---
    title: "Article B"
    slug: "article-b"
    date: "2026-04-01"
    coverImage: "/images/journal/article-b.jpg"
    category: "language"
    summary: "Summary of article B"
    readingTime: "3 min read"
    ---
    Article B body text.`;

    describe('getAllArticles', () => {
      beforeEach(() => {
        vi.mocked(fs.existsSync).mockReturnValue(true);
        vi.mocked(fs.readdirSync).mockReturnValue(['article-a.mdx', 'article-b.mdx'] as any);
        vi.mocked(fs.readFileSync).mockImplementation((p: any) => {
          if (String(p).includes('article-a')) return FIXTURE_A;
          if (String(p).includes('article-b')) return FIXTURE_B;
          return '';
        });
      });
      afterEach(() => vi.clearAllMocks());

      it('returns both articles sorted newest-first', () => {
        const articles = getAllArticles();
        expect(articles).toHaveLength(2);
        expect(articles[0].slug).toBe('article-a'); // 2026-05-10 > 2026-04-01
        expect(articles[1].slug).toBe('article-b');
      });

      it('computes readingTimeComputed from content when not in frontmatter', () => {
        const articles = getAllArticles();
        const a = articles.find((x) => x.slug === 'article-a')!;
        expect(a.readingTimeComputed).toMatch(/\d+ min read/);
      });

      it('uses frontmatter readingTime when present', () => {
        const articles = getAllArticles();
        const b = articles.find((x) => x.slug === 'article-b')!;
        expect(b.readingTimeComputed).toBe('3 min read');
      });

      it('returns [] when content directory does not exist', () => {
        vi.mocked(fs.existsSync).mockReturnValue(false);
        expect(getAllArticles()).toEqual([]);
      });
    });

    describe('getArticleSource', () => {
      beforeEach(() => {
        vi.mocked(fs.existsSync).mockReturnValue(true);
        vi.mocked(fs.readFileSync).mockReturnValue(FIXTURE_A);
      });
      afterEach(() => vi.clearAllMocks());

      it('returns source string for a valid slug', () => {
        const src = getArticleSource('magikarp-legend');
        expect(src).toBe(FIXTURE_A);
      });

      it('returns null for an unknown slug', () => {
        vi.mocked(fs.existsSync).mockReturnValue(false);
        expect(getArticleSource('no-such-article')).toBeNull();
      });

      it('returns null for path traversal slug ../../etc/passwd', () => {
        expect(getArticleSource('../../etc/passwd')).toBeNull();
      });

      it('returns null for slug with forward slash', () => {
        expect(getArticleSource('bad/slug')).toBeNull();
      });
    });
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| URL slug → filesystem | Slug from URL params passed to getArticleSource(); path traversal possible if unvalidated |
| MDX content → rendered HTML | Subutai-authored MDX compiled by next-mdx-remote/rsc; not user input |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14.6-01 | Information Disclosure | getArticleSource() slug → fs.readFileSync | mitigate | `SAFE_SLUG_RE = /^[a-z0-9-]+$/` guard before path.join; returns null on mismatch — implemented in Task 1 |
| T-14.6-02 | Tampering | MDX arbitrary JSX execution | accept | Content authored by Subutai (developer-controlled), not user-submitted; next-mdx-remote/rsc blocks raw JS execution in RSC mode |
| T-14.6-03 | Tampering | `import 'server-only'` enforcement | mitigate | Hard build error if articles.ts accidentally bundled client-side; fs module unavailable in browser bundle would surface the error |
</threat_model>
