**`src/app/[locale]/journal/page.tsx` (locale-ordered journal listing — D-12):**
This is the most important locale-specific behavior. Create `src/app/[locale]/journal/page.tsx`:
```typescript
import { getArticlesByLocale } from '@/lib/journal/articles';
import { getTranslations } from 'next-intl/server';
import { setRequestLocale } from 'next-intl/server';
import { LanguageBadge } from '@/components/ui/LanguageBadge';
// ... other existing journal page imports ...

export default async function LocaleJournalPage({ params }: { params: Promise<{ locale: string }> }) {
  const { locale } = await params;
  setRequestLocale(locale);
  
  const t = await getTranslations('journal');
  
  // D-12: locale articles first, then EN articles below with EN badge
  const localeArticles = locale !== 'en' ? getArticlesByLocale(locale as 'pt-BR' | 'es') : [];
  const enArticles = getArticlesByLocale('en');
  
  return (
    <main>
      <h1>{t('heading')}</h1>
      <p>{t('subheading')}</p>
      
      {/* Locale articles appear first (D-12) */}
      {localeArticles.map(article => (
        <JournalCard key={article.slug} article={article} locale={locale} />
        // No badge on locale articles — they are the primary content
      ))}
      
      {/* EN articles below with EN badge (D-12) */}
      {enArticles.map(article => (
        <JournalCard
          key={article.slug}
          article={article}
          locale="en"
          badge={locale !== 'en' ? <LanguageBadge locale="en" /> : undefined}
        />
      ))}
    </main>
  );
}
```

Read the current `src/app/journal/page.tsx` and use its exact card/list component structure — only modify the data sourcing (locale articles first) and add LanguageBadge.
