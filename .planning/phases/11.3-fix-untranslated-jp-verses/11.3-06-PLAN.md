    ```typescript
    /**
     * validate-review-entry.ts — Validate a single hand-drafted replace_verse_number
     * entry against the LIVE VerseSchema. Used by the Plan 06 inline reviewer.
     *
     * Usage:
     *   npx tsx scripts/seed/validate-review-entry.ts <slug> <verse_number>
     *
     * Reads .planning/verse-patches/<slug>.json, finds the replace_verse_number
     * entry with the given verse_number, validates it against VerseSchema, exits 0
     * on success or 1 on failure (printing the Zod error path).
     */

    import { readFileSync } from "fs";
    import { join, resolve } from "path";
    import { fileURLToPath } from "url";
    import { VerseSchema } from "../types/lesson.js";

    const ROOT = resolve(fileURLToPath(new URL(".", import.meta.url)), "../..");
    const PATCHES_DIR = join(ROOT, ".planning/verse-patches");

    const [, , slug, vnArg] = process.argv;
    if (!slug || !vnArg) {
      console.error("usage: validate-review-entry.ts <slug> <verse_number>");
      process.exit(2);
    }
    const verseNumber = parseInt(vnArg, 10);

    const patchPath = join(PATCHES_DIR, `${slug}.json`);
    const patch = JSON.parse(readFileSync(patchPath, "utf-8")) as {
      patches: Array<{ replace_verse_number?: number; after_original?: number; verse: any }>;
    };

    const entry = patch.patches.find(
      (p) => "replace_verse_number" in p && p.replace_verse_number === verseNumber
    );
    if (!entry) {
      console.error(`no replace_verse_number=${verseNumber} entry in ${patchPath}`);
      process.exit(1);
    }

    // Stamp verse_number to satisfy VerseSchema's int().min(1) constraint.
    const candidate = { verse_number: verseNumber, ...entry.verse };
    const v = VerseSchema.safeParse(candidate);
    if (!v.success) {
      console.error(`schema fail at ${slug}#${verseNumber}: ${v.error.errors.map((e) => `${e.path.join('.')}=${e.message}`).join('; ')}`);
      process.exit(1);
    }

    // Token-level meaning sanity (reuses Plan 05 D-08 triggers c+d so the reviewer
    // can't accidentally re-introduce a flagged shape):
    for (const t of v.data.tokens) {
      const en = t.meaning?.["en"] ?? "";
      if (en.includes("?") || /\bunknown\b/i.test(en) || en.trim() === "") {
        console.error(`token meaning.en is suspicious for surface=${t.surface}: "${en}"`);
        process.exit(1);
      }
    }
    for (const locale of ["en", "pt-BR", "es"] as const) {
      if ((v.data.translations[locale] ?? "").startsWith("(")) {
        console.error(`translations.${locale} regressed to stub signature`);
        process.exit(1);
      }
    }

    console.log(`[ok] ${slug}#${verseNumber} validates against VerseSchema and passes D-08 triggers`);
    ```

    The reviewer in Task 2 calls this after each entry edit.
