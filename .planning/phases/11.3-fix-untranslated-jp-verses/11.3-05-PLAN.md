    export async function callOllama(prompt: string): Promise<{ response: string; latency_ms: number }> {
      if (LOCKED_OLLAMA_MODEL === "PILOT_PENDING") {
        throw new Error("Plan 04 pilot must run and lock a model before drafter executes");
      }
      const t0 = performance.now();
      const res = await fetch(OLLAMA_ENDPOINT, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          model: LOCKED_OLLAMA_MODEL,
          prompt,
          stream: false,
          format: "json",
        }),
      });
      if (!res.ok) {
        const body = await res.text();
        throw new Error(`Ollama HTTP ${res.status}: ${body.slice(0, 500)}`);
      }
      const data = (await res.json()) as { response: string };
      return { response: data.response, latency_ms: performance.now() - t0 };
    }

    **Export 3: `detectFlagReason(parsed)` D-08 trigger detector:**

    ```typescript
    import { VerseSchema } from "../types/lesson.js";

    export type FlagReason =
      | "json_parse_fail"
      | "schema_fail"
      | "meaning_question_mark"
      | "meaning_unknown"
      | "meaning_empty"
      | "translation_stub_regression";

    export function detectFlagReason(rawResponse: string): FlagReason | { ok: true; verse: unknown } {
      let parsed: unknown;
      try { parsed = JSON.parse(rawResponse); } catch { return "json_parse_fail"; }

      // Stamp verse_number=0 so VerseSchema (which requires verse_number>=1) doesn't fail on its absence.
      // We re-validate verse_number-bearing structure later in the drafter.
      const candidate = { verse_number: 1, ...(parsed as object) };
      const v = VerseSchema.safeParse(candidate);
      if (!v.success) return "schema_fail";

      const verse = v.data;

      // D-08 (c) — token-level meaning.en signal scan
      for (const t of verse.tokens) {
        const en = t.meaning?.["en"] ?? "";
        if (en.includes("?")) return "meaning_question_mark";
        if (/\bunknown\b/i.test(en)) return "meaning_unknown";
        if (en.trim().length === 0) return "meaning_empty";
      }

      // D-08 (d) — verse-level translation regressing to stub signature
      for (const locale of ["en", "pt-BR", "es"] as const) {
        const t = verse.translations?.[locale] ?? "";
        if (typeof t === "string" && t.startsWith("(")) return "translation_stub_regression";
      }

      return { ok: true, verse: parsed }; // return the parsed-but-unstamped object so caller controls verse_number
    }
    ```
