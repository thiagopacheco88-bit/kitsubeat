- Document the choice inline; pick the simpler one (lesson's last verse end_time_ms is fine for 95%+ of the catalog).

**Per-row check:**
```ts
function auditLesson(slug: string, lesson: Lesson): Flag[] {
  const flags: Flag[] = [];
  const tvDurMs = lesson.verses.at(-1)?.end_time_ms ?? 0;
  const tvDurSec = tvDurMs / 1000;
  if (tvDurSec > 0) {
    const density = lesson.verses.length / tvDurSec;
    if (density < DENSITY_FLOOR) flags.push({ slug, kind: "density", density, threshold: DENSITY_FLOOR });
  } else {
    flags.push({ slug, kind: "no-tv-duration" });  // or alt fallback
  }
  for (const v of lesson.verses) {
    const span = v.end_time_ms - v.start_time_ms;
    if (span > MAX_VERSE_SPAN_MS) flags.push({ slug, kind: "max-span", verseNumber: v.verse_number, span_ms: span, threshold: MAX_VERSE_SPAN_MS });
  }
  return flags;
}
```

```ts
/**
 * snapshot-tv-lessons.ts — Pre-rollout snapshot of song_versions.lesson for version_type='tv'.
 *
 * Per D-09 (CONTEXT.md): writes to .planning/phases/11.2-.../tv-lessons-pre-rework-snapshot.json,
 * which IS committed to git (the data/ gitignore does not cover .planning/).
 *
 * Run BEFORE Plan 11.2-07's directory swap + 10c-load-tv-lessons.ts invocation.
 */

import { db } from "@/lib/db";
import { songVersions } from "@/lib/db/schema";
import { eq } from "drizzle-orm";
import * as fs from "node:fs";
import * as path from "node:path";

const SNAPSHOT_PATH = path.resolve(
  ".planning/phases/11.2-tv-derive-rework-demucs-nw/tv-lessons-pre-rework-snapshot.json"
);

async function main() {
  const rows = await db
    .select({ slug: songVersions.slug, lesson: songVersions.lesson })
    .from(songVersions)
    .where(eq(songVersions.versionType, "tv"));

  // Validate: 60 rows expected (per SPEC.md target)
  console.log(`[snapshot] queried ${rows.length} TV rows`);
  fs.writeFileSync(SNAPSHOT_PATH, JSON.stringify(rows, null, 2));
  console.log(`[snapshot] wrote ${SNAPSHOT_PATH}`);
}
main().catch(e => { console.error(e); process.exit(1); });
```
